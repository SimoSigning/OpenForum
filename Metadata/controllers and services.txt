using EntityModels;
using Repository.interfaces;

namespace Application.Services.Forum
{
    public class TopicService : ITopicService
    {
        private readonly ITopicRepository topicRepository;
        private readonly ITopicMapper topicMapper;
        private readonly ICurrentUserService currentUser;
        private readonly IUserService userService;
        private const int PageSize = 5;

        public TopicService(
            ITopicRepository topicRepository,
            ITopicMapper topicMapper,
            ICurrentUserService currentUser,
            IUserService userService)
        {
            this.topicRepository = topicRepository;
            this.topicMapper = topicMapper;
            this.currentUser = currentUser;
            this.userService = userService;
        }

        public async Task<TopicIndexView> GetTopicIndexViewAsync(int page = 1, TopicCreateView? createTopic = null)
        {
            page = Math.Max(1, page);

            (IEnumerable<Topic> Items, int TotalCount) result = await topicRepository.GetPageAsync(page, PageSize);

            int totalPages = Math.Max(1, (int)Math.Ceiling(result.TotalCount / (double)PageSize));

            // If the requested page is beyond the last page,
            // use the last valid page instead.
            if (page > totalPages)
            {
                page = totalPages;
                result = await topicRepository.GetPageAsync(page, PageSize);
            }

            string? userId = currentUser.GetUserIdOrNull();

            List<TopicListItemView> topics = result.Items.Select(
                topic => 
                {
                    TopicListItemView item = topicMapper.ToListItemView(topic);
                    item.CreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId, topic.CreatedByUserId, item.Creator);
                    item.CanManage = userId != null && topic.CreatedByUserId == userId;
                    return item;
                })
                .ToList();

            return new TopicIndexView
            {
                Topics = topics,
                CreateTopic = createTopic ?? new TopicCreateView(),
                EditTopic = null,
                CanCreate = userId != null,
                CurrentPage = page,
                TotalPages = totalPages
            };
        }

        public async Task<TopicIndexView?> GetEditViewAsync(int id, int page = 1, TopicEditView? editTopic = null)
        {
            Topic? topic = await topicRepository.GetByIdAsync(id);

            if (topic == null)
                return null;

            string userId = currentUser.GetRequiredUserId();

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this topic.");

            TopicIndexView viewModel = await GetTopicIndexViewAsync(page);

            viewModel.EditTopic = editTopic ?? topicMapper.ToEditView(topic);

            return viewModel;
        }

        public async Task CreateAsync(TopicCreateView model)
        {
            string userId = currentUser.GetRequiredUserId();

            Topic topic = topicMapper.ToEntity(model);
            topic.CreatedByUserId = userId;
            topic.CreatedByUserName = await userService.GetUsernameByIdAsync(userId);

            await topicRepository.AddAsync(topic);
        }

        public async Task UpdateAsync(TopicEditView model)
        {
            string userId = currentUser.GetRequiredUserId();
            Topic? topic = await topicRepository.GetByIdAsync(model.Id);

            if (topic == null)
                return;

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this topic.");

            topic.Title = model.Title;

            await topicRepository.UpdateAsync(topic);
        }

        public async Task DeleteAsync(int id)
        {
            string userId = currentUser.GetRequiredUserId();

            Topic? topic = await topicRepository.GetByIdAsync(id);

            if (topic == null)
                return;

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to delete this topic.");

            await topicRepository.DeleteAsync(id);
        }

    }

    public class PostService : IPostService
    {
        private readonly IPostRepository postRepository;
        private readonly ITopicRepository topicRepository;
        private readonly ILikeRepository likeRepository;
        private readonly IPostMapper postMapper;
        private readonly ICurrentUserService currentUser;
        private readonly IUserService userService;
        private const int PageSize = 5;

        public PostService(
            IPostRepository postRepository,
            ITopicRepository topicRepository,
            ILikeRepository likeRepository,
            IPostMapper postMapper,
            ICurrentUserService currentUser,
            IUserService userService)
        {
            this.postRepository = postRepository;
            this.topicRepository = topicRepository;
            this.likeRepository = likeRepository;
            this.postMapper = postMapper;
            this.currentUser = currentUser;
            this.userService = userService;
        }

        public async Task<PostIndexView?> GetPostIndexViewAsync(int topicId, int page = 1, PostCreateView? createPost = null)
        {
            Topic? topic = await topicRepository.GetByIdAsync(topicId);

            if (topic == null)
                return null;

            page = Math.Max(1, page);

            (IEnumerable<Post> Items, int TotalCount) result = await postRepository.GetPageByTopicIdAsync(topicId, page, PageSize);

            int totalPages = Math.Max(1, (int)Math.Ceiling(result.TotalCount / (double)PageSize));

            // If the requested page is beyond the last page,
            // use the last valid page instead.
            if (page > totalPages)
            {
                page = totalPages;
                result = await postRepository.GetPageByTopicIdAsync(topicId, page, PageSize);
            }

            string? userId = currentUser.GetUserIdOrNull();

            List<Post> postItems = result.Items.ToList();
            List<int> postIds = postItems.Select(p => p.Id).ToList();

            Dictionary<int, int> likeCounts = await likeRepository.GetPostLikeCountsAsync(postIds);
            HashSet<int> likedPostIds = userId == null ? new HashSet<int>() : (await likeRepository.GetPostIdsLikedByUserAsync(postIds, userId)).ToHashSet();

            List<PostListItemView> posts = postItems.Select(post =>
            {
                PostListItemView item = postMapper.ToListItemView(post);
                item.CreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId, post.CreatedByUserId, item.Creator);
                item.CanManage = userId != null && post.CreatedByUserId == userId;
                item.LikeCount = likeCounts.GetValueOrDefault(post.Id);
                item.IsLikedByCurrentUser = likedPostIds.Contains(post.Id);
                item.CanLike = userId != null;
                return item;
            }).ToList();

            return new PostIndexView
            {
                TopicId = topic.Id,
                TopicTitle = topic.Title,
                TopicCreator = topic.CreatedByUserName,
                TopicCreatorId = topic.CreatedByUserId,
                TopicCreatedAt = topic.CreatedAt,
                Posts = posts,
                CreatePost = createPost ?? new PostCreateView
                {
                    TopicId = topicId
                },
                EditPost = null,
                CanCreate = userId != null,
                CurrentPage = page,
                TotalPages = totalPages
            };
        }

        public async Task<PostIndexView?> GetEditViewAsync(int postId, int page = 1, PostEditView? editPost = null)
        {
            Post? post = await postRepository.GetByIdAsync(postId);

            if (post == null)
                return null;

            string userId = currentUser.GetRequiredUserId();

            if (post.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this post.");

            PostIndexView? viewModel = await GetPostIndexViewAsync(post.TopicId, page);

            if (viewModel == null)
                return null;

            viewModel.EditPost = editPost ?? postMapper.ToEditView(post);

            return viewModel;
        }

        public async Task CreateAsync(PostCreateView model)
        {
            string userId = currentUser.GetRequiredUserId();

            Post post = postMapper.ToEntity(model);
            post.CreatedByUserId = userId;
            post.CreatedByUserName = await userService.GetUsernameByIdAsync(userId);

        public async Task UpdateAsync(PostEditView model)

            string userId = currentUser.GetRequiredUserId();
            Post? post = await postRepository.GetByIdAsync(model.Id);

            if (post == null)
                return;

            if (post.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this post.");

            post.Id = model.Id;
            post.Content = model.Content;

            await postRepository.UpdateAsync(post);
        }

        public async Task DeleteAsync(int id)
        {
            string userId = currentUser.GetRequiredUserId();
            Post? post = await postRepository.GetByIdAsync(id);

            if (post == null)
                return;

            if (post.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to delete this post.");

            await likeRepository.DeleteByPostIdAsync(id);
            await postRepository.DeleteAsync(id);
        }

    }

namespace OpenForum.Controllers

    public class CommentController : Controller
    {
        private readonly ICommentService commentService;

        public CommentController(ICommentService commentService)
        {
            this.commentService = commentService;
        }

        [HttpGet]
        public async Task<IActionResult> Details(int id, int page = 1)
        {
            CommentIndexView? model = await commentService.GetCommentIndexViewAsync(id, page);

            if (model == null)
                return NotFound();

            ViewData["CurrentPage"] = model.CurrentPage;

            return View(model);
        }

        [Authorize]
        [HttpGet]
        public async Task<IActionResult> Edit(int id, int page = 1)
        {
            try
            {
                CommentIndexView? model = await commentService.GetEditViewAsync(id, page);

                if (model == null)
                    return NotFound();

                ViewData["CurrentPage"] = model.CurrentPage;

                return View("Details", model);
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Create(CommentCreateView model, int page = 1)
        {

            // Only top-level comments need a title.
            if (!model.ParentCommentId.HasValue && string.IsNullOrWhiteSpace(model.Title))
            {
                ModelState.AddModelError(nameof(model.Title), "A title is required for a comment.");
            }

            if (!ModelState.IsValid)
            {
                CommentIndexView? invalidView = await commentService.GetCommentIndexViewAsync(model.PostId, page, model);

                if (invalidView == null)
                    return NotFound();

                ViewData["CurrentPage"] = invalidView.CurrentPage;
                ViewData["PostId"] = model.PostId;

                return View("Details", invalidView);
            }

            try
            {
                await commentService.CreateAsync(model);

                if (model.ParentCommentId.HasValue)
                {
                    return Redirect(Url.Action(nameof(Details), "Comment", new { id = model.PostId, page }) + $"#comment-{model.ParentCommentId.Value}");
                }

                return RedirectToAction(nameof(Details), new { id = model.PostId, page });
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
            catch (ArgumentException ex)
            {
                ModelState.AddModelError(string.Empty, ex.Message);

                CommentIndexView? invalidView = await commentService.GetCommentIndexViewAsync(model.PostId, page, model);

                if (invalidView == null)
                    return NotFound();

                ViewData["CurrentPage"] = invalidView.CurrentPage;
                ViewData["PostId"] = model.PostId;

                return View("Details", invalidView);
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Update(CommentEditView model, int page = 1)
        {
            if (!ModelState.IsValid)
            {
                try
                {
                    CommentIndexView? viewModel = await commentService.GetEditViewAsync(model.Id, page, model);

                    if (viewModel == null)
                        return NotFound();

                    ViewData["CurrentPage"] = viewModel.CurrentPage;

                    return View("Details", viewModel);
                }
                catch (UnauthorizedAccessException)
                {
                    return Forbid();
                }
            }

            try
            {
                await commentService.UpdateAsync(model);

                if (model.ParentCommentId.HasValue)
                {
                    return Redirect(Url.Action(nameof(Details), "Comment", new { id = model.PostId, page }) + $"#comment-{model.ParentCommentId.Value}");
                }

                return RedirectToAction(nameof(Details), new { id = model.PostId, page });
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]

using Application.common.Application.Common;
using Application.Interfaces.Contexts;
using Application.Interfaces.Mappers.Forum;
using Application.Interfaces.Services.Forum;
using Application.Interfaces.Services.User;
using EntityModels;
using Microsoft.EntityFrameworkCore.Metadata.Internal;
using Repository.interfaces;
using ViewModels.Comment;

namespace Application.Services.Forum
{
    public class CommentService : ICommentService
    {
        private readonly ICommentRepository commentRepository;
        private readonly IPostRepository postRepository;
        private readonly ILikeRepository likeRepository;
        private readonly ICommentMapper commentMapper;
        private readonly ICurrentUserService currentUser;
        private readonly IUserService userService;
        private const int PageSize = 5;

        public CommentService(
            ICommentRepository commentRepository,
            IPostRepository postRepository,
            ILikeRepository likeRepository,
            ICommentMapper commentMapper,
            ICurrentUserService currentUser,
            IUserService userService
            )
        {
            this.commentRepository = commentRepository;
            this.postRepository = postRepository;
            this.likeRepository = likeRepository;
            this.commentMapper = commentMapper;
            this.currentUser = currentUser;
            this.userService = userService;
        }


        public async Task<CommentIndexView?> GetCommentIndexViewAsync(int postId, int page = 1, CommentCreateView? createComment = null)
        {
            Post? post = await postRepository.GetByIdAsync(postId);

            if (post == null)
                return null;

            page = Math.Max(1, page);

            (IEnumerable<Comment> Items, int TotalCount) result = await commentRepository.GetPageByPostIdAsync(postId, page, PageSize);

            int totalPages = (int)Math.Ceiling(result.TotalCount / (double)PageSize);

            if (totalPages > 0 && page > totalPages)
            {
                page = totalPages;
                result = await commentRepository.GetPageByPostIdAsync(postId, page, PageSize);
            }

            string? userId = currentUser.GetUserIdOrNull();

            Dictionary<int, CommentListItemView> commentsById = result.Items.ToDictionary(
                c => c.Id,
                c =>
                {
                    CommentListItemView view = commentMapper.ToListItemView(c);
                    view.CreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId, c.CreatedByUserId, view.Creator);
                    view.CanManage = userId != null && c.CreatedByUserId == userId;
                    return view;
                });

            List<int> parentIds = result.Items.Select(c => c.Id).ToList();

            while (parentIds.Count > 0)
            {
                List<Comment> replies = await commentRepository.GetRepliesByParentIdsAsync(postId, parentIds);
                parentIds = new List<int>();

                foreach (Comment reply in replies)
                {
                    CommentListItemView replyView = commentMapper.ToListItemView(reply);
                    replyView.CreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId,reply.CreatedByUserId, replyView.Creator);
                    replyView.CanManage = reply.CreatedByUserId == userId;

                    if (reply.ParentCommentId.HasValue && commentsById.TryGetValue(reply.ParentCommentId.Value, out CommentListItemView? parentView))
                    {
                        replyView.ParentTitle = !string.IsNullOrWhiteSpace(parentView.Title) ? parentView.Title : parentView.Content;
                        replyView.ParentCreator = parentView.Creator;
                        parentView.Replies.Add(replyView);
                    }

                    commentsById.Add(reply.Id, replyView);
                    parentIds.Add(reply.Id);
                }
            }

            int postLikeCount = await likeRepository.GetPostLikeCountAsync(post.Id);
            bool isPostLikedByCurrentUser = userId != null && await likeRepository.IsPostLikedByUserAsync(post.Id, userId);
            List<int> allCommentIds = commentsById.Keys.ToList();
            Dictionary<int, int> likeCounts = await likeRepository.GetCommentLikeCountsAsync(allCommentIds);
            HashSet<int> likedCommentIds = userId == null ? 
                new HashSet<int>() : (await likeRepository.GetCommentIdsLikedByUserAsync(allCommentIds, userId)).ToHashSet();

            foreach (KeyValuePair<int, CommentListItemView> entry in commentsById)
            {
                CommentListItemView view = entry.Value;
                view.LikeCount = likeCounts.GetValueOrDefault(entry.Key);
                view.IsLikedByCurrentUser = likedCommentIds.Contains(entry.Key);
                view.CanLike = userId != null;
            }

            List<CommentListItemView> comments = result.Items.Select(c => commentsById[c.Id]).ToList();

            return new CommentIndexView
            {
                PostId = post.Id,
                TopicId = post.TopicId,
                PostTitle = post.Title,
                PostContent = post.Content,
                PostCreator = post.CreatedByUserName,
                PostCreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId, post.CreatedByUserId, post.CreatedByUserName),
                PostCreatorId = post.CreatedByUserId,
                PostCreatedAt = post.CreatedAt,
                Comments = comments,
                CreateComment = createComment ?? new CommentCreateView
                {
                    PostId = postId
                },
                CanCreate = !string.IsNullOrEmpty(userId),
                EditComment = null,
                CurrentPage = page,
                TotalPages = totalPages,
                PostLikeCount = postLikeCount,
                IsPostLikedByCurrentUser = isPostLikedByCurrentUser,
                CanLikePost = userId != null,
            };
        }


        public async Task<CommentIndexView?> GetEditViewAsync(int commentId, int page = 1, CommentEditView? editComment = null)
        {
            Comment? comment = await commentRepository.GetByIdAsync(commentId);

            if (comment == null)
                return null;

            string userId = currentUser.GetRequiredUserId();

            if (comment.CreatedByUserId != userId)
            {
                throw new UnauthorizedAccessException("You are not allowed to edit this comment.");
            }

            Post? post = await postRepository.GetByIdAsync(comment.PostId);

            if (post == null)
                return null;

            CommentIndexView? viewModel = await GetCommentIndexViewAsync(post.Id, page);

            if (viewModel == null)
                return null;

            viewModel.EditComment = editComment ?? commentMapper.ToEditView(comment);

            return viewModel;
        }


        public async Task CreateAsync(CommentCreateView model)
        {
            string userId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(model.Content))
            {
                throw new ArgumentException("Comment content is required.");
            }

            if (!model.ParentCommentId.HasValue &&
                string.IsNullOrWhiteSpace(model.Title))
            {
                throw new ArgumentException("A title is required for a top-level comment.");
            }

            if (model.ParentCommentId.HasValue)
            {
                Comment? parent = await commentRepository.GetByIdAsync(model.ParentCommentId.Value);

                if (parent == null || parent.PostId != model.PostId)
                {
                    throw new ArgumentException("The parent comment does not belong to this post.");
                }
            }

            Comment comment = commentMapper.ToEntity(model);
            comment.CreatedByUserId = userId;
            comment.CreatedByUserName = await userService.GetUsernameByIdAsync(userId);

            await commentRepository.AddAsync(comment);
        }



        public async Task UpdateAsync(CommentEditView model)
        {
            string userId = currentUser.GetRequiredUserId();
            Comment? comment = await commentRepository.GetByIdAsync(model.Id);

            if (comment == null)
                return;

            if (comment.CreatedByUserId != userId)
            {
                throw new UnauthorizedAccessException("You are not allowed to edit this comment.");
            }

            comment.Title = comment.ParentCommentId.HasValue ? string.Empty : model.Title?.Trim() ?? string.Empty;
            comment.Content = model.Content;
            await commentRepository.UpdateAsync(comment);
        }

        public async Task DeleteAsync(int id)
        {
            string userId = currentUser.GetRequiredUserId();
            Comment? comment = await commentRepository.GetByIdAsync(id);

            if (comment == null)
                return;

            if (comment.CreatedByUserId != userId)
            {
                throw new UnauthorizedAccessException("You are not allowed to delete this comment.");
            }

            List<Comment> comments = (await commentRepository.GetByPostIdAsync(comment.PostId)).ToList();

            Dictionary<int, Comment> commentsById = comments.ToDictionary(c => c.Id);

            Dictionary<int, List<Comment>> repliesByParent = comments
                .Where(c => c.ParentCommentId.HasValue)
                .GroupBy(c => c.ParentCommentId!.Value)
                .ToDictionary(g => g.Key, g => g.ToList());

            List<Comment> deletionOrder = new();

            void AddCommentAndReplies(int commentId)
            {
                if (!commentsById.TryGetValue(commentId, out Comment? current))
                    return;

                deletionOrder.Add(current);

                if (repliesByParent.TryGetValue(commentId, out List<Comment>? replies))
                {
                    foreach (Comment reply in replies)
                    {
                        AddCommentAndReplies(reply.Id);
                    }
                }
            }

            
            AddCommentAndReplies(comment.Id);

            // Mark descendants before their parents for deletion.
            deletionOrder.Reverse();

            // Delete likes belonging to the comment and all its replies.
            List<int> commentIdsToDelete = deletionOrder.Select(c => c.Id).ToList();

            await likeRepository.DeleteByCommentIdsAsync(commentIdsToDelete);
            await commentRepository.DeleteRangeAsync(deletionOrder);
        }

    }
}

                isLiked = true;

using Application.Interfaces.Services.Features;
using Application.Interfaces.Services.User;
using Application.Models;
using EntityModels;
using Repository.interfaces;
using System;
using System.Collections.Generic;
using System.Text;
using ViewModels.Friends;

namespace Application.Services.Features
{
    public class FriendshipService : IFriendshipService
    {
        private readonly IFriendshipRepository friendshipRepository;
        private readonly IUserService userService;
        private readonly ICurrentUserService currentUserService;

        private const int PageSize = 10;

        public FriendshipService(
            IFriendshipRepository friendshipRepository,
            IUserService userService,
            ICurrentUserService currentUserService)
        {
            this.friendshipRepository = friendshipRepository;
            this.userService = userService;
            this.currentUserService = currentUserService;
        }

        public async Task SendRequestAsync(string addresseeUserId)
        {
            string requesterUserId = currentUserService.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(addresseeUserId))
                throw new ArgumentException("A recipient user ID is required.");

            if (requesterUserId == addresseeUserId)
                throw new InvalidOperationException("You cannot send a friend request to yourself.");

            bool recipientExists = await userService.UserExistsAsync(addresseeUserId);

            if (!recipientExists)
                throw new KeyNotFoundException("The requested user does not exist.");

            Friendship? existingFriendship = await friendshipRepository.GetBetweenUsersAsync(requesterUserId, addresseeUserId);

            if (existingFriendship != null)
                throw new InvalidOperationException("A friendship or friend request already exists between these users.");

            // Normalize the pair so the unique database index works
            // regardless of which user sends the request.
            string userIdLow;
            string userIdHigh;

            if (string.Compare(requesterUserId, addresseeUserId, StringComparison.Ordinal) < 0)
            {
                userIdLow = requesterUserId;
                userIdHigh = addresseeUserId;
            }
            else
            {
                userIdLow = addresseeUserId;
                userIdHigh = requesterUserId;
            }

            Friendship friendship = new Friendship
            {
                RequesterUserId = requesterUserId,
                AddresseeUserId = addresseeUserId,
                UserIdLow = userIdLow,
                UserIdHigh = userIdHigh,
                Status = FriendshipStatus.Pending,
                CreatedAt = DateTime.UtcNow
            };

            await friendshipRepository.AddAsync(friendship);
        }

        public async Task AcceptRequestAsync(int friendshipId)
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            Friendship friendship = await GetFriendshipOrThrowAsync(friendshipId);

            if (friendship.AddresseeUserId != currentUserId)
                throw new UnauthorizedAccessException("Only the recipient can accept this friend request.");

            if (friendship.Status != FriendshipStatus.Pending)
                throw new InvalidOperationException("This friend request is no longer pending.");

            friendship.Status = FriendshipStatus.Accepted;

            await friendshipRepository.UpdateAsync(friendship);
        }

        public async Task DeclineRequestAsync(int friendshipId)
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            Friendship friendship = await GetFriendshipOrThrowAsync(friendshipId);

            if (friendship.AddresseeUserId != currentUserId)
                throw new UnauthorizedAccessException("Only the recipient can decline this friend request.");

            if (friendship.Status != FriendshipStatus.Pending)
                throw new InvalidOperationException("This friend request is no longer pending.");

            await friendshipRepository.DeleteAsync(friendship.Id);
        }

        public async Task RemoveFriendAsync(int friendshipId)
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            Friendship friendship = await GetFriendshipOrThrowAsync(friendshipId);

            bool isParticipant = friendship.RequesterUserId == currentUserId || friendship.AddresseeUserId == currentUserId;

            if (!isParticipant)
                throw new UnauthorizedAccessException("You are not part of this friendship.");

            if (friendship.Status != FriendshipStatus.Accepted)
                throw new InvalidOperationException("Only accepted friendships can be removed here.");

            await friendshipRepository.DeleteAsync(friendship.Id);
        }

        public async Task<IEnumerable<Friendship>> GetFriendsAsync()
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            return await friendshipRepository.GetAcceptedForUserAsync(currentUserId);
        }

        public async Task<IEnumerable<Friendship>> GetIncomingRequestsAsync()
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            return await friendshipRepository.GetIncomingPendingRequestsAsync(currentUserId);
        }

        private async Task<Friendship> GetFriendshipOrThrowAsync(int id)
        {
            Friendship? friendship = await friendshipRepository.GetByIdAsync(id);

            if (friendship == null)
                throw new KeyNotFoundException("Friendship or friend request not found.");

            return friendship;
        }

        public async Task<UserSearchView> SearchUsersAsync(string query,int page = 1)
        {
            page = Math.Max(1, page);

            string currentUserId = currentUserService.GetRequiredUserId();

            query = query?.Trim() ?? string.Empty;

            if (string.IsNullOrWhiteSpace(query))
            {
                return new UserSearchView
                {
                    SearchQuery = query,
                    Users = Enumerable.Empty<UserSearchItemView>(),
                    CurrentPage = 1,
                    TotalPages = 1
                };
            }

            int skip = (page - 1) * PageSize;

            (IEnumerable<UserSearchResult> Items, int TotalCount) searchResult = await userService.SearchUsersByUsernameAsync(query, currentUserId, skip, PageSize);

            int totalPages = Math.Max(1, (int)Math.Ceiling(searchResult.TotalCount / (double)PageSize));

            // Correct the page if it is beyond the last available page.
            if (page > totalPages)
            {
                page = totalPages;
                skip = (page - 1) * PageSize;
                searchResult = await userService.SearchUsersByUsernameAsync(query, currentUserId, skip, PageSize);
            }

            IEnumerable<Friendship> relationships = await friendshipRepository.GetAllForUserAsync(currentUserId);

            Dictionary<string, Friendship> relationshipLookup = relationships.ToDictionary(f => f.RequesterUserId == currentUserId ? f.AddresseeUserId : f.RequesterUserId);

            List<UserSearchItemView> users = searchResult.Items.Select(user =>
                {
                    relationshipLookup.TryGetValue(user.UserId, out Friendship? relationship);

                    string? status = null;
                    bool canSendRequest = relationship == null;

                    if (relationship != null)
                    {
                        if (relationship.Status == FriendshipStatus.Accepted)
                        {
                            status = "Friends";
                        }
                        else if (relationship.RequesterUserId == currentUserId)
                        {
                            status = "Request sent";
                        }
                        else
                        {
                            status = "Request received";
                        }
                    }

                    return new UserSearchItemView
                    {
                        UserId = user.UserId,
                        Username = user.Username,
                        CanSendRequest = canSendRequest,
                        RelationshipStatus = status
                    };
                })
                .ToList();

            return new UserSearchView
            {
                SearchQuery = query,
                Users = users,
                CurrentPage = page,
                TotalPages = totalPages
            };
        }

        public async Task<FriendsIndexView> GetFriendsIndexViewAsync()
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            // Retrieve accepted friendships and incoming requests.
            List<Friendship> friendships = (await GetFriendsAsync()).ToList();

            List<Friendship> incomingRequests = (await GetIncomingRequestsAsync()).ToList();

            // Find the IDs of all other users.
            List<string> userIds = friendships
                .Select(f => f.RequesterUserId == currentUserId ? f.AddresseeUserId : f.RequesterUserId)
                .Concat(incomingRequests.Select(f => f.RequesterUserId))
                .Distinct()
                .ToList();

            // Retrieve all usernames in one call.
            Dictionary<string, string> usernames = await userService.GetUsernamesByIdsAsync(userIds);

            // Map accepted friendships.
            List<UserSearchItemView> friends = friendships
                .Select(f =>
                {
                    string friendUserId = f.RequesterUserId == currentUserId ? f.AddresseeUserId : f.RequesterUserId;

                    usernames.TryGetValue(friendUserId, out string? username);

                    return new UserSearchItemView
                    {
                        UserId = friendUserId,
                        FriendshipId = f.Id,
                        Username = username ?? "Unknown user",
                        CanSendRequest = false,
                        RelationshipStatus = "Friends"
                    };
                })
                .ToList();

            // Map incoming requests.
            List<FriendRequestItemView> requests = incomingRequests
                .Select(f =>
                {
                    usernames.TryGetValue(f.RequesterUserId, out string? username);

                    return new FriendRequestItemView
                    {
                        FriendshipId = f.Id,
                        UserId = f.RequesterUserId,
                        Username = username ?? "Unknown user",
                        CreatedAt = f.CreatedAt
                    };
                })
                .ToList();

            return new FriendsIndexView
            {
                Friends = friends,
                IncomingRequests = requests
            };
        }

        public async Task<UserRelationshipView> GetRelationshipAsync(string targetUserId)
        {
            string currentUserId = currentUserService.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(targetUserId))
                throw new ArgumentException("A target user ID is required.");

            // A user cannot send a friend request to themselves.
            if (currentUserId == targetUserId)
            {
                return new UserRelationshipView
                {
                    CanSendRequest = false
                };
            }

            Friendship? relationship = await friendshipRepository.GetBetweenUsersAsync(currentUserId,targetUserId);

            if (relationship == null)
            {
                return new UserRelationshipView
                {
                    CanSendRequest = true
                };
            }

            return new UserRelationshipView
            {
                FriendshipId = relationship.Id,
                CanSendRequest = false,
                RelationshipStatus = relationship.Status switch
                {
                    FriendshipStatus.Accepted => "Friends",
                    FriendshipStatus.Pending when relationship.RequesterUserId == currentUserId => "Request sent",
                    FriendshipStatus.Pending => "Request received",
                    _ => null
                }
            };
        }
    }

namespace Application.Services.Features

    public class MessageService : IMessageService
    {
        private readonly IMessageRepository messageRepository;
        private readonly IUserService userService;
        private readonly ICurrentUserService currentUser;

        public MessageService(
            IMessageRepository messageRepository,
            IUserService userService,
            ICurrentUserService currentUser)
        {
            this.messageRepository = messageRepository;
            this.userService = userService;
            this.currentUser = currentUser;
        }

        public async Task<InboxView> GetInboxAsync()
        {
            string userId = currentUser.GetRequiredUserId();

            List<Message> messages =
                await messageRepository.GetInboxAsync(userId);

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(
                    messages.Select(m => m.SenderUserId));

            return new InboxView
            {
                Messages = messages.Select(m => new MessageItemView
                {
                    Id = m.Id,
                    OtherUserId = m.SenderUserId,
                    OtherUserName = usernames.GetValueOrDefault(
                        m.SenderUserId, "Unknown user"),
                    Subject = m.Subject,
                    BodyPreview = CreatePreview(m.Body),
                    SentAt = m.SentAt,
                    IsRead = m.IsRead
                }),

                UnreadCount = await messageRepository.GetUnreadCountAsync(userId)
            };
        }

        public async Task<SentMessagesView> GetSentAsync()
        {
            string userId = currentUser.GetRequiredUserId();

            List<Message> messages =
                await messageRepository.GetSentAsync(userId);

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(
                    messages.Select(m => m.RecipientUserId));

            return new SentMessagesView
            {
                Messages = messages.Select(m => new MessageItemView
                {
                    Id = m.Id,
                    OtherUserId = m.RecipientUserId,
                    OtherUserName = usernames.GetValueOrDefault(
                        m.RecipientUserId, "Unknown user"),
                    Subject = m.Subject,
                    BodyPreview = CreatePreview(m.Body),
                    SentAt = m.SentAt,
                    IsRead = m.IsRead
                })
            };
        }

        public async Task<MessageDetailsView> GetDetailsAsync(int messageId)
        {
            string userId = currentUser.GetRequiredUserId();

            Message? message =
                await messageRepository.GetByIdAsync(messageId);

            if (message == null)
            {
                throw new KeyNotFoundException("Message not found.");
            }

            bool isRecipient = message.RecipientUserId == userId;
            bool isSender = message.SenderUserId == userId;

            if (!isRecipient && !isSender)
            {
                throw new UnauthorizedAccessException(
                    "You are not allowed to access this message.");
            }

            // Only mark incoming messages as read.
            if (isRecipient && !message.IsRead)
            {
                message.IsRead = true;

                await messageRepository.UpdateAsync(message);
            }

            string senderUsername =
                await userService.GetUsernameByIdAsync(message.SenderUserId);

            string recipientUsername =
                await userService.GetUsernameByIdAsync(message.RecipientUserId);

            return new MessageDetailsView
            {
                Id = message.Id,

                SenderUserId = message.SenderUserId,
                SenderUserName = senderUsername,

                RecipientUserId = message.RecipientUserId,
                RecipientUserName = recipientUsername,

                Subject = message.Subject,
                Body = message.Body,
                SentAt = message.SentAt,
                IsRead = message.IsRead,

                CanReply = isRecipient
            };
        }

        public async Task<ComposeMessageView> GetComposeViewAsync(
            string recipientUserName)
        {
            string currentUserId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(recipientUserName))
            {
                throw new ArgumentException("A recipient is required.");
            }

            string recipientUserId =
                await userService.GetUserIdByUsernameAsync(recipientUserName);

            if (recipientUserId == currentUserId)
            {
                throw new ArgumentException(
                    "You cannot send a message to yourself.");
            }

            string username =
                await userService.GetUsernameByIdAsync(recipientUserId);

            return new ComposeMessageView
            {
                RecipientUserId = recipientUserId,
                RecipientUserName = username
            };
        }

        public async Task<ComposeMessageView> GetReplyComposeViewAsync(
            int originalMessageId)
        {
            string userId = currentUser.GetRequiredUserId();

            Message? originalMessage =
                await messageRepository.GetByIdAsync(originalMessageId);

            if (originalMessage == null)
            {
                throw new KeyNotFoundException("Original message not found.");
            }

            // Only the recipient of the original message can reply.
            if (originalMessage.RecipientUserId != userId)
            {
                throw new UnauthorizedAccessException(
                    "You are not allowed to reply to this message.");
            }

            if (!await userService.UserExistsAsync(originalMessage.SenderUserId))
            {
                throw new InvalidOperationException(
                    "The original sender no longer exists.");
            }

            string username =
                await userService.GetUsernameByIdAsync(
                    originalMessage.SenderUserId);

            return new ComposeMessageView
            {
                RecipientUserId = originalMessage.SenderUserId,
                RecipientUserName = username,
                Subject = CreateReplySubject(originalMessage.Subject),
                IsReply = true,
                OriginalMessageId = originalMessage.Id
            };
        }

        public async Task SendAsync(ComposeMessageView model)
        {
            string senderUserId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(model.Body))
            {
                throw new ArgumentException("Message body is required.");
            }

            string recipientUserId;
            string subject;

            if (model.IsReply)
            {
                if (!model.OriginalMessageId.HasValue)
                {
                    throw new ArgumentException("Original message is required for a reply.");
                }

                Message? originalMessage =
                    await messageRepository.GetByIdAsync(model.OriginalMessageId.Value);

                if (originalMessage == null)
                {
                    throw new KeyNotFoundException(
                        "Original message not found.");
                }

                if (originalMessage.RecipientUserId != senderUserId)
                {
                    throw new UnauthorizedAccessException(
                        "You are not allowed to reply to this message.");
                }

                recipientUserId = originalMessage.SenderUserId;

                // Never trust the submitted reply subject.
                subject = CreateReplySubject(originalMessage.Subject);
            }
            else
            {
                if (string.IsNullOrWhiteSpace(model.RecipientUserName))
                {
                    throw new ArgumentException("A recipient is required.");
                }

                // Resolve the recipient again instead of trusting a hidden ID.
                recipientUserId =
                    await userService.GetUserIdByUsernameAsync(
                        model.RecipientUserName);

                if (recipientUserId == senderUserId)
                {
                    throw new ArgumentException(
                        "You cannot send a message to yourself.");
                }

                subject = model.Subject?.Trim() ?? string.Empty;

                if (string.IsNullOrWhiteSpace(subject))
                {
                    throw new ArgumentException("Subject is required.");
                }
            }

            if (recipientUserId == senderUserId)
            {
                throw new ArgumentException(
                    "You cannot send a message to yourself.");
            }

            if (!await userService.UserExistsAsync(recipientUserId))
            {
                throw new KeyNotFoundException("Recipient not found.");
            }

            string body = model.Body.Trim();

            if (body.Length > 5000)
            {
                throw new ArgumentException(
                    "Message body cannot exceed 5000 characters.");
            }

            if (subject.Length > 150)
            {
                throw new ArgumentException(
                    "Message subject cannot exceed 150 characters.");
            }

            Message message = new Message
            {
                SenderUserId = senderUserId,
                RecipientUserId = recipientUserId,
                Subject = subject,
                Body = body,
                SentAt = DateTime.UtcNow,
                IsRead = false
            };

            await messageRepository.AddAsync(message);
        }

        public async Task<int> GetUnreadCountAsync()
        {
            string userId = currentUser.GetRequiredUserId();

            return await messageRepository.GetUnreadCountAsync(userId);
        }

        private static string CreatePreview(string body)
        {
            if (string.IsNullOrWhiteSpace(body))
            {
                return string.Empty;
            }

            const int previewLength = 100;

            string preview = body.Trim();

            if (preview.Length <= previewLength)
            {
                return preview;
            }

            return preview.Substring(0, previewLength) + "...";
        }

        private static string CreateReplySubject(string originalSubject)
        {
            const string prefix = "Reply to: ";

            string subject = prefix + originalSubject;

            // The database allows a maximum of 150 characters.
            return subject.Length <= 150
                ? subject
                : subject.Substring(0, 150);
        }
    }

using Application.Interfaces.Services.Features;
using Application.Interfaces.Services.User;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using ViewModels.Friends;
using ViewModels.PublicUserProfile;

namespace OpenForum.Controllers
{
    public class PublicProfileController : Controller
    {
        private readonly IUserProfileService profileService;
        private readonly IFriendshipService friendshipService;
        private readonly IUserService userService;

        public PublicProfileController(
            IUserProfileService profileService,
            IFriendshipService friendshipService,
            IUserService userService)
        {
            this.profileService = profileService;
            this.friendshipService = friendshipService;
            this.userService = userService;
        }

        [Authorize]
        [HttpGet]
        public async Task<IActionResult> Index(
            string userName,
            string? returnUrl)
        {
            if (string.IsNullOrWhiteSpace(userName))
            {
                return BadRequest();
            }

            try
            {
                // Resolve the ID internally using the username.
                string userId =
                    await userService.GetUserIdByUsernameAsync(userName);

                PublicUserProfileView profileView =
                    await profileService.GetPublicProfileAsync(
                        userId, userName);

                UserRelationshipView relationship =
                    await friendshipService.GetRelationshipAsync(
                        profileView.UserId);

                profileView.RelationshipStatus =
                    relationship.RelationshipStatus;

                profileView.CanSendFriendRequest =
                    relationship.CanSendRequest;

                profileView.FriendshipId =
                    relationship.FriendshipId;

                ViewData["ReturnUrl"] = Url.IsLocalUrl(returnUrl)
                    ? returnUrl
                    : Url.Action("Index", "Topic");

                return View(profileView);
            }
            catch (KeyNotFoundException)
            {
                return NotFound();
            }
        }
    }
