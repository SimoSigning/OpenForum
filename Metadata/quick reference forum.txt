
namespace EntityModels
{
    public class Topic
    {
        public int Id { get; set; }

        public string Title { get; set; } = string.Empty;

        public string CreatedByUserName { get; set; } = string.Empty;

        public string CreatedByUserId { get; set; } = string.Empty;

        public DateTime CreatedAt { get; set; } = DateTime.UtcNow;

    }
}



using EntityModels;
using Microsoft.EntityFrameworkCore;

namespace EntityContext
{
    public class ForumContext : DbContext
    {
        public ForumContext(DbContextOptions<ForumContext> options) : base(options){}

        public DbSet<Topic> Topics => Set<Topic>();
        public DbSet<Post> Posts => Set<Post>();
        public DbSet<Comment> Comments => Set<Comment>();
        public DbSet<Like> Likes => Set<Like>();
        public DbSet<UserProfile> UserProfiles => Set<UserProfile>();
        public DbSet<Friendship> Friendships => Set<Friendship>();
        public DbSet<Message> Messages => Set<Message>();
        public DbSet<ChatRoom> ChatRooms => Set<ChatRoom>();
        public DbSet<ChatRoomMember> ChatRoomMembers => Set<ChatRoomMember>();
        public DbSet<ChatMessage> ChatMessages => Set<ChatMessage>();

        protected override void OnModelCreating(ModelBuilder modelBuilder)
        {
            base.OnModelCreating(modelBuilder);

            modelBuilder.Entity<Post>()
                .HasOne(p => p.Topic)
                .WithMany()
                .HasForeignKey(p => p.TopicId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Comment>()
                .HasOne(c => c.Post)
                .WithMany(p => p.Comments)
                .HasForeignKey(c => c.PostId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Comment>()
                .HasOne(c => c.ParentComment)
                .WithMany(c => c.Replies)
                .HasForeignKey(c => c.ParentCommentId)
                .OnDelete(DeleteBehavior.NoAction);

            modelBuilder.Entity<Like>(entity =>
            {
                entity.ToTable("Likes", table =>
                    table.HasCheckConstraint(
                        "CK_Likes_ExactlyOneTarget",
                        """
                        ([PostId] IS NOT NULL AND [CommentId] IS NULL)
                        OR
                        ([PostId] IS NULL AND [CommentId] IS NOT NULL)
                        """
                        ));

                entity.HasKey(l => l.Id);

                entity.Property(l => l.UserId)
                    .IsRequired();

                entity.Property(l => l.CreatedAt)
                    .IsRequired();

                entity.HasOne(l => l.Post)
                    .WithMany(p => p.Likes)
                    .HasForeignKey(l => l.PostId)
                    .OnDelete(DeleteBehavior.Restrict);

                entity.HasOne(l => l.Comment)
                    .WithMany(c => c.Likes)
                    .HasForeignKey(l => l.CommentId)
                    .OnDelete(DeleteBehavior.Restrict);

                entity.HasIndex(l => new { l.UserId, l.PostId })
                    .IsUnique()
                    .HasFilter("[PostId] IS NOT NULL");

                entity.HasIndex(l => new { l.UserId, l.CommentId })
                    .IsUnique()
                    .HasFilter("[CommentId] IS NOT NULL");
            });

            modelBuilder.Entity<UserProfile>(entity =>
            {
                entity.HasKey(p => p.Id);

                entity.Property(p => p.UserId)
                    .IsRequired();

                entity.HasIndex(p => p.UserId)
                    .IsUnique();

                entity.Property(p => p.FirstName)
                    .HasMaxLength(100);

                entity.Property(p => p.LastName)
                    .HasMaxLength(100);

                entity.Property(p => p.Address)
                    .HasMaxLength(200);

                entity.Property(p => p.PostalCode)
                    .HasMaxLength(20);

                entity.Property(p => p.City)
                    .HasMaxLength(100);

                entity.Property(p => p.ProfileText)
                    .HasMaxLength(5000);
            });

            modelBuilder.Entity<Friendship>(entity =>
            {
                entity.ToTable("Friendships", table =>
                    table.HasCheckConstraint(
                        "CK_Friendships_DifferentUsers",
                        "[UserIdLow] <> [UserIdHigh]"
                    ));

                entity.Property(f => f.RequesterUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.AddresseeUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.UserIdLow)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.UserIdHigh)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.HasIndex(f => new
                {
                    f.UserIdLow,
                    f.UserIdHigh
                }).IsUnique();
            });

            modelBuilder.Entity<Message>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.SenderUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.RecipientUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.Subject)
                    .IsRequired()
                    .HasMaxLength(150);

                entity.Property(m => m.Body)
                    .IsRequired()
                    .HasMaxLength(5000);

                entity.Property(m => m.SentAt)
                    .IsRequired();

                entity.Property(m => m.IsRead)
                    .IsRequired();

                // Helps retrieve inbox messages and unread messages.
                entity.HasIndex(m => new
                {
                    m.RecipientUserId,
                    m.IsRead,
                    m.SentAt
                });

                // Helps retrieve sent messages.
                entity.HasIndex(m => new
                {
                    m.SenderUserId,
                    m.SentAt
                });
            });

            // Chat rooms
            modelBuilder.Entity<ChatRoom>(entity =>
            {
                entity.HasKey(r => r.Id);

                entity.Property(r => r.Name)
                    .IsRequired()
                    .HasMaxLength(100);

                entity.Property(r => r.RoomType)
                    .IsRequired();

                entity.Property(r => r.CreatedByUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(r => r.OwnerUserId)
                    .HasMaxLength(450);

                entity.Property(r => r.CreatedAt)
                    .IsRequired();

                // A topic can only have one chatroom.
                entity.HasIndex(r => r.TopicId)
                    .IsUnique()
                    .HasFilter("[TopicId] IS NOT NULL");

                // Topic relationship.
                entity.HasOne(r => r.Topic)
                    .WithOne()
                    .HasForeignKey<ChatRoom>(r => r.TopicId)
                    .OnDelete(DeleteBehavior.Cascade);

                // Helpful when retrieving rooms by type.
                entity.HasIndex(r => r.RoomType);
            });

            // Chat room members
            modelBuilder.Entity<ChatRoomMember>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.UserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.JoinedAt)
                    .IsRequired();

                entity.HasOne(m => m.ChatRoom)
                    .WithMany(r => r.Members)
                    .HasForeignKey(m => m.ChatRoomId)
                    .OnDelete(DeleteBehavior.Cascade);

                // A user can only belong to a room once.
                entity.HasIndex(m => new
                {
                    m.ChatRoomId,
                    m.UserId
                }).IsUnique();

                // Find all rooms belonging to a user.
                entity.HasIndex(m => m.UserId);
            });

            // Chat messages
            modelBuilder.Entity<ChatMessage>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.SenderUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.Content)
                    .IsRequired()
                    .HasMaxLength(2000);

                entity.Property(m => m.SentAt)
                    .IsRequired();

                entity.HasOne(m => m.ChatRoom)
                    .WithMany(r => r.Messages)
                    .HasForeignKey(m => m.ChatRoomId)
                    .OnDelete(DeleteBehavior.Cascade);

                // Supports retrieving the latest messages efficiently.
                entity.HasIndex(m => new
                {
                    m.ChatRoomId,
                    m.Id
                });
            });

            SeedData(modelBuilder);
        }
        private static void SeedData(ModelBuilder modelBuilder)
        {
            int topicId = 1;
            int postId = 1;
            int commentId = 1;

            modelBuilder.Entity<Topic>().HasData(
                new Topic
                {
                    Id = topicId,
                    Title = "Welcome to the forum",
                    CreatedByUserId = "seed-user",
                    CreatedAt = new DateTime(2025, 1, 1, 0, 0, 0, DateTimeKind.Utc)
                }
            );

            modelBuilder.Entity<Post>().HasData(
                new Post
                {
                    Id = postId,
                    TopicId = topicId,
                    Content = "This is the first post",
                    CreatedByUserId = "seed-user",
                    CreatedAt = new DateTime(2025, 1, 1, 0, 0, 0, DateTimeKind.Utc)
                }
            );

            modelBuilder.Entity<Comment>().HasData(
                new Comment
                {
                    Id = commentId,
                    PostId = postId,
                    Content = "First comment!",
                    CreatedByUserId = "seed-user",
                    CreatedAt = new DateTime(2025, 1, 1, 0, 0, 0, DateTimeKind.Utc)
                }
            );
        }
    }
}


using Microsoft.EntityFrameworkCore;
using EntityModels;
using Repository.interfaces;
using EntityContext;

namespace Repository
{
    public class TopicRepository : ITopicRepository
    {
        private readonly ForumContext context;

        public TopicRepository(ForumContext forumContext)
        {
            context = forumContext;
        }

        public async Task<(IEnumerable<Topic> Items, int TotalCount)> GetPageAsync(int page, int pageSize)
        {
            if (page < 1)
                throw new ArgumentOutOfRangeException(nameof(page));

            if (pageSize < 1)
                throw new ArgumentOutOfRangeException(nameof(pageSize));

            int totalCount = await context.Topics.CountAsync();

            List<Topic> items = await context.Topics
                .OrderByDescending(t => t.CreatedAt)
                .ThenByDescending(t => t.Id)
                .Skip((page - 1) * pageSize)
                .Take(pageSize)
                .ToListAsync();

            return (items, totalCount);
        }

        public async Task<IEnumerable<Topic>> GetByCreatedByUserIdAsync(string userId, int skip, int take)
        {
            return await context.Topics
                .AsNoTracking()
                .Where(t => t.CreatedByUserId == userId)
                .OrderByDescending(t => t.CreatedAt)
                .ThenByDescending(t => t.Id)
                .Skip(skip)
                .Take(take)
                .ToListAsync();
        }

        public async Task<int> CountByCreatedByUserIdAsync(string userId)
        {
            return await context.Topics.CountAsync(t => t.CreatedByUserId == userId);
        }

        public async Task<Topic?> GetByIdAsync(int id)
        {
            return await context.Topics.FirstOrDefaultAsync(t => t.Id == id);
        }

        public async Task<IEnumerable<Topic>> SearchByTitleAsync(string query)
        {
            return await context.Topics
                .AsNoTracking()
                .Where(t => t.Title.Contains(query))
                .OrderByDescending(t => t.CreatedAt)
                .ThenByDescending(t => t.Id)
                .Take(20)
                .ToListAsync();
        }

        public async Task<IEnumerable<Topic>> GetPopularTopicsAsync()
        {
            return await context.Topics
                .AsNoTracking()
                .OrderByDescending(t => context.Posts.Count(p => p.TopicId == t.Id))
                .ThenByDescending(t => t.CreatedAt)
                .ThenByDescending(t => t.Id)
                .Take(10)
                .ToListAsync();
        }

        public async Task AddAsync(Topic topic)
        {
            context.Topics.Add(topic);

            await context.SaveChangesAsync();
        }

        public async Task UpdateAsync(Topic topic)
        {
            context.Topics.Update(topic);

            await context.SaveChangesAsync();
        }

        public async Task DeleteAsync(int id)
        {
            Topic? topic = await context.Topics.FindAsync(id);

            if (topic == null)
                return;

            context.Topics.Remove(topic);

            await context.SaveChangesAsync();
        }
    }
}



using Application.common.Application.Common;
using Application.Interfaces.Contexts;
using Application.Interfaces.Mappers.Forum;
using Application.Interfaces.Services.Forum;
using Application.Interfaces.Services.User;
using EntityModels;
using Repository.interfaces;
using ViewModels.Topic;

namespace Application.Services.Forum
{
    public class TopicService : ITopicService
    {
        private readonly ITopicRepository topicRepository;
        private readonly ITopicMapper topicMapper;
        private readonly ICurrentUserService currentUser;
        private readonly IUserService userService;
        private const int PageSize = 5;

        public TopicService(
            ITopicRepository topicRepository,
            ITopicMapper topicMapper,
            ICurrentUserService currentUser,
            IUserService userService)
        {
            this.topicRepository = topicRepository;
            this.topicMapper = topicMapper;
            this.currentUser = currentUser;
            this.userService = userService;
        }

        public async Task<TopicIndexView> GetTopicIndexViewAsync(int page = 1, TopicCreateView? createTopic = null)
        {
            page = Math.Max(1, page);

            (IEnumerable<Topic> Items, int TotalCount) result = await topicRepository.GetPageAsync(page, PageSize);

            int totalPages = Math.Max(1, (int)Math.Ceiling(result.TotalCount / (double)PageSize));

            // If the requested page is beyond the last page,
            // use the last valid page instead.
            if (page > totalPages)
            {
                page = totalPages;
                result = await topicRepository.GetPageAsync(page, PageSize);
            }

            string? userId = currentUser.GetUserIdOrNull();

            List<TopicListItemView> topics = result.Items.Select(
                topic => 
                {
                    TopicListItemView item = topicMapper.ToListItemView(topic);
                    item.CreatorDisplayName = CreatorDisplayNameResolver.Resolve(userId, topic.CreatedByUserId, item.Creator);
                    item.CanManage = userId != null && topic.CreatedByUserId == userId;
                    return item;
                })
                .ToList();

            return new TopicIndexView
            {
                Topics = topics,
                CreateTopic = createTopic ?? new TopicCreateView(),
                EditTopic = null,
                CanCreate = userId != null,
                CurrentPage = page,
                TotalPages = totalPages
            };
        }

        public async Task<TopicIndexView?> GetEditViewAsync(int id, int page = 1, TopicEditView? editTopic = null)
        {
            Topic? topic = await topicRepository.GetByIdAsync(id);

            if (topic == null)
                return null;

            string userId = currentUser.GetRequiredUserId();

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this topic.");

            TopicIndexView viewModel = await GetTopicIndexViewAsync(page);

            viewModel.EditTopic = editTopic ?? topicMapper.ToEditView(topic);

            return viewModel;
        }

        public async Task CreateAsync(TopicCreateView model)
        {
            string userId = currentUser.GetRequiredUserId();

            Topic topic = topicMapper.ToEntity(model);
            topic.CreatedByUserId = userId;
            topic.CreatedByUserName = await userService.GetUsernameByIdAsync(userId);

            await topicRepository.AddAsync(topic);
        }

        public async Task UpdateAsync(TopicEditView model)
        {
            string userId = currentUser.GetRequiredUserId();
            Topic? topic = await topicRepository.GetByIdAsync(model.Id);

            if (topic == null)
                return;

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to edit this topic.");

            topic.Title = model.Title;

            await topicRepository.UpdateAsync(topic);
        }

        public async Task DeleteAsync(int id)
        {
            string userId = currentUser.GetRequiredUserId();

            Topic? topic = await topicRepository.GetByIdAsync(id);

            if (topic == null)
                return;

            if (topic.CreatedByUserId != userId)
                throw new UnauthorizedAccessException("You are not allowed to delete this topic.");

            await topicRepository.DeleteAsync(id);
        }

    }
}





using Application.Interfaces.Services.Forum;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using ViewModels.Topic;

namespace OpenForum.Controllers
{
    public class TopicController : Controller
    {
        private readonly ITopicService topicService;

        public TopicController(ITopicService topicService)
        {
            this.topicService = topicService;
        }

        [HttpGet]
        public async Task<IActionResult> Index(int page = 1)
        {
            TopicIndexView model = await topicService.GetTopicIndexViewAsync(page);

            return View(model);
        }

        [Authorize]
        [HttpGet]
        public async Task<IActionResult> Edit(int id, int page = 1)
        {
            try
            {
                TopicIndexView? model = await topicService.GetEditViewAsync(id, page);

                if (model == null)
                    return NotFound();

                return View("Index", model);
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Create(TopicCreateView model, int page = 1)
        {
            if (!ModelState.IsValid)
            {
                TopicIndexView viewModel = await topicService.GetTopicIndexViewAsync(page, model);

                return View("Index", viewModel);
            }

            try
            {
                await topicService.CreateAsync(model);

                return RedirectToAction(nameof(Index), new { page });
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Update(TopicEditView model, int page = 1)
        {
            if (!ModelState.IsValid)
            {
                try
                {
                    TopicIndexView? viewModel = await topicService.GetEditViewAsync(model.Id, page, model);

                    if (viewModel == null)
                        return NotFound();

                    return View("Index", viewModel);
                }
                catch (UnauthorizedAccessException)
                {
                    return Forbid();
                }
            }

            try
            {
                await topicService.UpdateAsync(model);

                return RedirectToAction(nameof(Index), new { page });
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }

        [Authorize]
        [HttpPost]
        [ValidateAntiForgeryToken]
        public async Task<IActionResult> Delete(int id, int page = 1)
        {
            try
            {
                await topicService.DeleteAsync(id);

                return RedirectToAction(nameof(Index), new { page });
            }
            catch (UnauthorizedAccessException)
            {
                return Forbid();
            }
        }
    }
