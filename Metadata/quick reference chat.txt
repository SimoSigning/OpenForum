using EntityModels;
using Microsoft.EntityFrameworkCore;

namespace EntityContext
{
    public class ForumContext : DbContext
    {
        public ForumContext(DbContextOptions<ForumContext> options) : base(options){}

        public DbSet<Topic> Topics => Set<Topic>();
        public DbSet<Post> Posts => Set<Post>();
        public DbSet<Comment> Comments => Set<Comment>();
        public DbSet<Like> Likes => Set<Like>();
        public DbSet<UserProfile> UserProfiles => Set<UserProfile>();
        public DbSet<Friendship> Friendships => Set<Friendship>();
        public DbSet<Message> Messages => Set<Message>();
        public DbSet<ChatRoom> ChatRooms => Set<ChatRoom>();
        public DbSet<ChatRoomMember> ChatRoomMembers => Set<ChatRoomMember>();
        public DbSet<ChatMessage> ChatMessages => Set<ChatMessage>();

        protected override void OnModelCreating(ModelBuilder modelBuilder)
        {
            base.OnModelCreating(modelBuilder);

            modelBuilder.Entity<Post>()
                .HasOne(p => p.Topic)
                .WithMany()
                .HasForeignKey(p => p.TopicId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Comment>()
                .HasOne(c => c.Post)
                .WithMany(p => p.Comments)
                .HasForeignKey(c => c.PostId)
                .OnDelete(DeleteBehavior.Cascade);

            modelBuilder.Entity<Comment>()
                .HasOne(c => c.ParentComment)
                .WithMany(c => c.Replies)
                .HasForeignKey(c => c.ParentCommentId)
                .OnDelete(DeleteBehavior.NoAction);

            modelBuilder.Entity<Like>(entity =>
            {
                entity.ToTable("Likes", table =>
                    table.HasCheckConstraint(
                        "CK_Likes_ExactlyOneTarget",
                        """
                        ([PostId] IS NOT NULL AND [CommentId] IS NULL)
                        OR
                        ([PostId] IS NULL AND [CommentId] IS NOT NULL)
                        """
                        ));

                entity.HasKey(l => l.Id);

                entity.Property(l => l.UserId)
                    .IsRequired();

                entity.Property(l => l.CreatedAt)
                    .IsRequired();

                entity.HasOne(l => l.Post)
                    .WithMany(p => p.Likes)
                    .HasForeignKey(l => l.PostId)
                    .OnDelete(DeleteBehavior.Restrict);

                entity.HasOne(l => l.Comment)
                    .WithMany(c => c.Likes)
                    .HasForeignKey(l => l.CommentId)
                    .OnDelete(DeleteBehavior.Restrict);

                entity.HasIndex(l => new { l.UserId, l.PostId })
                    .IsUnique()
                    .HasFilter("[PostId] IS NOT NULL");

                entity.HasIndex(l => new { l.UserId, l.CommentId })
                    .IsUnique()
                    .HasFilter("[CommentId] IS NOT NULL");
            });

            modelBuilder.Entity<UserProfile>(entity =>
            {
                entity.HasKey(p => p.Id);

                entity.Property(p => p.UserId)
                    .IsRequired();

                entity.HasIndex(p => p.UserId)
                    .IsUnique();

                entity.Property(p => p.FirstName)
                    .HasMaxLength(100);

                entity.Property(p => p.LastName)
                    .HasMaxLength(100);

                entity.Property(p => p.Address)
                    .HasMaxLength(200);

                entity.Property(p => p.PostalCode)
                    .HasMaxLength(20);

                entity.Property(p => p.City)
                    .HasMaxLength(100);

                entity.Property(p => p.ProfileText)
                    .HasMaxLength(5000);
            });

            modelBuilder.Entity<Friendship>(entity =>
            {
                entity.ToTable("Friendships", table =>
                    table.HasCheckConstraint(
                        "CK_Friendships_DifferentUsers",
                        "[UserIdLow] <> [UserIdHigh]"
                    ));

                entity.Property(f => f.RequesterUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.AddresseeUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.UserIdLow)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(f => f.UserIdHigh)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.HasIndex(f => new
                {
                    f.UserIdLow,
                    f.UserIdHigh
                }).IsUnique();
            });

            modelBuilder.Entity<Message>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.SenderUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.RecipientUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.Subject)
                    .IsRequired()
                    .HasMaxLength(150);

                entity.Property(m => m.Body)
                    .IsRequired()
                    .HasMaxLength(5000);

                entity.Property(m => m.SentAt)
                    .IsRequired();

                entity.Property(m => m.IsRead)
                    .IsRequired();

                // Helps retrieve inbox messages and unread messages.
                entity.HasIndex(m => new
                {
                    m.RecipientUserId,
                    m.IsRead,
                    m.SentAt
                });

                // Helps retrieve sent messages.
                entity.HasIndex(m => new
                {
                    m.SenderUserId,
                    m.SentAt
                });
            });

            // Chat rooms
            modelBuilder.Entity<ChatRoom>(entity =>
            {
                entity.ToTable("ChatRooms", table =>
                table.HasCheckConstraint(
                    "CK_ChatRooms_ValidRoomConfiguration",
                    """
                    (
                        [RoomType] = 1
                        AND [TopicId] IS NULL
                        AND [OwnerUserId] IS NOT NULL
                    )
                    OR
                    (
                        [RoomType] = 2
                        AND [TopicId] IS NULL
                        AND [OwnerUserId] IS NULL
                    )
                    OR
                    (
                        [RoomType] = 3
                        AND [TopicId] IS NOT NULL
                        AND [OwnerUserId] IS NULL
                    )
                    """
                ));

                entity.HasKey(r => r.Id);

                entity.Property(r => r.Name)
                    .IsRequired()
                    .HasMaxLength(100);

                entity.Property(r => r.RoomType)
                    .IsRequired();

                entity.Property(r => r.CreatedByUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(r => r.OwnerUserId)
                    .HasMaxLength(450);

                entity.Property(r => r.CreatedAt)
                    .IsRequired();

                // A topic can only have one chatroom.
                entity.HasIndex(r => r.TopicId)
                    .IsUnique()
                    .HasFilter("[TopicId] IS NOT NULL");

                // Topic relationship.
                entity.HasOne(r => r.Topic)
                    .WithOne()
                    .HasForeignKey<ChatRoom>(r => r.TopicId)
                    .OnDelete(DeleteBehavior.Cascade);

                // Helpful when retrieving rooms by type.
                entity.HasIndex(r => r.RoomType);
            });



            // Chat room members
            modelBuilder.Entity<ChatRoomMember>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.UserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.JoinedAt)
                    .IsRequired();

                entity.HasOne(m => m.ChatRoom)
                    .WithMany(r => r.Members)
                    .HasForeignKey(m => m.ChatRoomId)
                    .OnDelete(DeleteBehavior.Cascade);

                // A user can only belong to a room once.
                entity.HasIndex(m => new
                {
                    m.ChatRoomId,
                    m.UserId
                }).IsUnique();

                // Find all rooms belonging to a user.
                entity.HasIndex(m => m.UserId);
            });

            // Chat messages
            modelBuilder.Entity<ChatMessage>(entity =>
            {
                entity.HasKey(m => m.Id);

                entity.Property(m => m.SenderUserId)
                    .IsRequired()
                    .HasMaxLength(450);

                entity.Property(m => m.Content)
                    .IsRequired()
                    .HasMaxLength(2000);

                entity.Property(m => m.SentAt)
                    .IsRequired();

                entity.HasOne(m => m.ChatRoom)
                    .WithMany(r => r.Messages)
                    .HasForeignKey(m => m.ChatRoomId)
                    .OnDelete(DeleteBehavior.Cascade);

                // Supports retrieving the latest messages efficiently.
                entity.HasIndex(m => new
                {
                    m.ChatRoomId,
                    m.Id
                });
            });

            SeedData(modelBuilder);
        }

using EntityContext;
using EntityModels;
using Microsoft.EntityFrameworkCore;
using Repository.interfaces;

namespace Repository
{
    public class ChatRepository : IChatRepository
    {
        private readonly ForumContext context;

        public ChatRepository(ForumContext forumContext)
        {
            context = forumContext;
        }

        // -------------------------
        // Chat rooms
        // -------------------------

        public async Task<ChatRoom?> GetRoomByIdAsync(int roomId)
        {
            return await context.ChatRooms.FirstOrDefaultAsync(r => r.Id == roomId);
        }

        public async Task<ChatRoom?> GetRoomByTopicIdAsync(int topicId)
        {
            return await context.ChatRooms.FirstOrDefaultAsync(r => r.TopicId == topicId);
        }

        public async Task<IEnumerable<ChatRoom>> GetRoomsForUserAsync(
            string userId)
        {
            return await context.ChatRooms
                .AsNoTracking()
                .Include(r => r.Members)
                .Where(r =>
                    r.RoomType == ChatRoomType.Private &&
                    r.Members.Any(m => m.UserId == userId))
                .OrderByDescending(r => r.CreatedAt)
                .ToListAsync();
        }

        public async Task<IEnumerable<ChatRoom>> GetMiscellaneousRoomsAsync()
        {
            return await context.ChatRooms
                .AsNoTracking()
                .Include(r => r.Members)
                .Where(r => r.RoomType == ChatRoomType.Miscellaneous)
                .OrderBy(r => r.Name)
                .ToListAsync();
        }

        public async Task AddRoomAsync(ChatRoom room)
        {
            context.ChatRooms.Add(room);

            await context.SaveChangesAsync();
        }

        public async Task DeleteRoomAsync(ChatRoom room)
        {
            context.ChatRooms.Remove(room);

            await context.SaveChangesAsync();
        }

        public async Task CreateRoomWithMembersAsync(
            ChatRoom room,
            IEnumerable<ChatRoomMember> members)
        {
            List<ChatRoomMember> memberList = members.ToList();

            bool ownsTransaction =
                context.Database.CurrentTransaction == null;

            await using var transaction = ownsTransaction
                ? await context.Database.BeginTransactionAsync()
                : null;

            try
            {
                context.ChatRooms.Add(room);

                foreach (ChatRoomMember member in memberList)
                {
                    room.Members.Add(member);
                }

                await context.SaveChangesAsync();

                if (ownsTransaction)
                {
                    await transaction!.CommitAsync();
                }
            }
            catch
            {
                try
                {
                    if (ownsTransaction && transaction != null)
                    {
                        await transaction.RollbackAsync();
                    }
                }
                finally
                {
                    // Remove the failed room and its members
                    // from EF Core's change tracker.
                    foreach (ChatRoomMember member in memberList)
                    {
                        context.Entry(member).State =
                            EntityState.Detached;
                    }

                    context.Entry(room).State =
                        EntityState.Detached;
                }

                throw;
            }
        }


        public async Task LeaveRoomAsync(int roomId, string userId)
        {
            await using var transaction =
                await context.Database.BeginTransactionAsync(
                    System.Data.IsolationLevel.Serializable);

            try
            {
                ChatRoom? room = await context.ChatRooms
                    .FirstOrDefaultAsync(r => r.Id == roomId);

                if (room == null)
                    throw new KeyNotFoundException("Chat room not found.");

                ChatRoomMember? member = await context.ChatRoomMembers
                    .FirstOrDefaultAsync(m =>
                        m.ChatRoomId == roomId &&
                        m.UserId == userId);

                if (member == null)
                    throw new InvalidOperationException(
                        "You are not a member of this chat room.");

                List<ChatRoomMember> remainingMembers =
                    await context.ChatRoomMembers
                        .Where(m =>
                            m.ChatRoomId == roomId &&
                            m.UserId != userId)
                        .ToListAsync();

                // Only private rooms are deleted when their final
                // member leaves.
                if (room.RoomType == ChatRoomType.Private &&
                    remainingMembers.Count == 0)
                {
                    context.ChatRooms.Remove(room);
                }
                else
                {
                    // Transfer ownership if the current owner leaves.
                    if (room.RoomType == ChatRoomType.Private &&
                        room.OwnerUserId == userId)
                    {
                        ChatRoomMember newOwner = remainingMembers
                            .OrderBy(m => m.JoinedAt)
                            .ThenBy(m => m.UserId, StringComparer.Ordinal)
                            .First();

                        room.OwnerUserId = newOwner.UserId;
                    }

                    context.ChatRoomMembers.Remove(member);
                }

                await context.SaveChangesAsync();

                await transaction.CommitAsync();
            }
            catch
            {
                await transaction.RollbackAsync();
                throw;
            }
        }

        //handling topics in chatrepository.. But its fine for now
        public async Task<string?> GetTopicTitleAsync(int topicId)
        {
            return await context.Topics
                .Where(t => t.Id == topicId)
                .Select(t => t.Title)
                .FirstOrDefaultAsync();
        }

        public async Task<bool> TopicExistsAsync(int topicId)
        {
            return await context.Topics
                .AnyAsync(t => t.Id == topicId);
        }

        // -------------------------
        // Membership
        // -------------------------

        public async Task<ChatRoomMember?> GetMembershipAsync(int roomId, string userId)
        {
            return await context.ChatRoomMembers
                .FirstOrDefaultAsync(m =>
                    m.ChatRoomId == roomId &&
                    m.UserId == userId);
        }

        public async Task<IEnumerable<ChatRoomMember>> GetMembersAsync(int roomId)
        {
            return await context.ChatRoomMembers
                .AsNoTracking()
                .Where(m => m.ChatRoomId == roomId)
                .OrderBy(m => m.JoinedAt)
                .ToListAsync();
        }

        public async Task AddMemberAsync(ChatRoomMember member)
        {
            context.ChatRoomMembers.Add(member);

            await context.SaveChangesAsync();
        }

        public async Task RemoveMemberAsync(ChatRoomMember member)
        {
            context.ChatRoomMembers.Remove(member);

            await context.SaveChangesAsync();
        }

        // -------------------------
        // Messages
        // -------------------------

        public async Task<IEnumerable<ChatMessage>> GetLatestMessagesAsync(int roomId, int count)
        {
            return await context.ChatMessages
                .AsNoTracking()
                .Where(m => m.ChatRoomId == roomId)
                .OrderByDescending(m => m.Id)
                .Take(count)
                .OrderBy(m => m.Id)
                .ToListAsync();
        }

        public async Task AddMessageAsync(ChatMessage message)
        {
            await using var transaction =
                await context.Database.BeginTransactionAsync();

            try
            {
                context.ChatMessages.Add(message);

                await context.SaveChangesAsync();

                List<long> oldMessageIds = await context.ChatMessages
                    .Where(m => m.ChatRoomId == message.ChatRoomId)
                    .OrderByDescending(m => m.Id)
                    .Skip(100)
                    .Select(m => m.Id)
                    .ToListAsync();

                if (oldMessageIds.Count > 0)
                {
                    await context.ChatMessages
                        .Where(m => oldMessageIds.Contains(m.Id))
                        .ExecuteDeleteAsync();
                }

                await transaction.CommitAsync();
            }
            catch
            {
                await transaction.RollbackAsync();
                throw;
            }
        }

        // -------------------------
        // Ownership
        // -------------------------

        public async Task UpdateRoomAsync(ChatRoom room)
        {
            context.ChatRooms.Update(room);

            await context.SaveChangesAsync();
        }
    }
}

using Application.Interfaces.Services.Features;
using Application.Interfaces.Services.User;

using System.Collections.Generic;

using ViewModels.Chat;

namespace Application.Services.Features
{
    public class ChatService : IChatService
    {
        private readonly IChatRepository chatRepository;
        private readonly IFriendshipRepository friendshipRepository;
        private readonly ICurrentUserService currentUser;
        private readonly IUserService userService;

        private const int MessageLimit = 100;
        private const int MaxMessageLength = 2000;

        public ChatService(
            IChatRepository chatRepository,
            IFriendshipRepository friendshipRepository,
            ICurrentUserService currentUser,
            IUserService userService)
        {
            this.chatRepository = chatRepository;
            this.friendshipRepository = friendshipRepository;
            this.currentUser = currentUser;
            this.userService = userService;
        }

        // --------------------------------
        // Room retrieval
        // --------------------------------

        public async Task<IEnumerable<ChatRoomView>> GetMyRoomsAsync()
        {
            string userId = currentUser.GetRequiredUserId();

            IEnumerable<ChatRoom> rooms =
                await chatRepository.GetRoomsForUserAsync(userId);

            return rooms.Select(room => MapRoom(room, userId));
        }

        public async Task<IEnumerable<ChatRoomView>> GetMiscellaneousRoomsAsync()
        {
            string userId = currentUser.GetRequiredUserId();

            IEnumerable<ChatRoom> rooms =
                await chatRepository.GetMiscellaneousRoomsAsync();

            return rooms.Select(room => new ChatRoomView
            {
                Id = room.Id,
                Name = room.Name,
                RoomType = room.RoomType.ToString(),
                OwnerUserId = room.OwnerUserId,
                IsOwner = false,
                IsMember = room.Members.Any(m => m.UserId == userId),
                MemberCount = room.Members.Count,
                TopicId = room.TopicId
            });
        }

        public async Task<ChatRoomView> GetRoomAsync(int roomId)
        {
            string userId = currentUser.GetRequiredUserId();

            ChatRoom room = await GetRoomOrThrowAsync(roomId);

            if (room.RoomType == ChatRoomType.Topic)
            {
                ChatRoomMember? membership =
                    await chatRepository.GetMembershipAsync(roomId, userId);

                if (membership == null)
                {
                    await chatRepository.AddMemberAsync(new ChatRoomMember
                    {
                        ChatRoomId = roomId,
                        UserId = userId,
                        JoinedAt = DateTime.UtcNow
                    });
                }
            }
            else if (room.RoomType == ChatRoomType.Miscellaneous)
            {
                await JoinMiscellaneousRoomAsync(roomId);
            }
            else if (room.RoomType == ChatRoomType.Private)
            {
                await EnsureMembershipAsync(roomId, userId);
            }
            else
            {
                throw new InvalidOperationException(
                    "Unsupported chat room type.");
            }

            IEnumerable<ChatRoomMember> members =
                await chatRepository.GetMembersAsync(room.Id);

            return new ChatRoomView
            {
                Id = room.Id,
                Name = room.Name,
                RoomType = room.RoomType.ToString(),
                OwnerUserId = room.OwnerUserId,
                IsOwner = room.OwnerUserId == userId,
                IsMember = true,
                MemberCount = members.Count(),
                TopicId = room.TopicId,
                CurrentUserId = userId
            };
        }

        public async Task<IEnumerable<ChatMemberView>> GetRoomMembersAsync(int roomId)
        {
            string currentUserId = currentUser.GetRequiredUserId();

            ChatRoom room = await GetRoomOrThrowAsync(roomId);

            // Verify that the current user belongs to the room.
            await EnsureMembershipAsync(roomId, currentUserId);

            IEnumerable<ChatRoomMember> members =
                await chatRepository.GetMembersAsync(roomId);

            List<ChatRoomMember> memberList = members.ToList();

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(
                    memberList.Select(m => m.UserId));

            return memberList.Select(member =>
                new ChatMemberView
                {
                    UserId = member.UserId,
                    Username = usernames.TryGetValue(
                        member.UserId,
                        out string? username)
                            ? username
                            : "Unknown user",
                    JoinedAt = member.JoinedAt,
                    IsOwner = room.OwnerUserId == member.UserId
                });
        }

        // --------------------------------
        // Private rooms
        // --------------------------------

        public async Task<IEnumerable<ChatFriendView>> GetFriendsForChatAsync()
        {
            string currentUserId = currentUser.GetRequiredUserId();

            IEnumerable<Friendship> friendships =
                await friendshipRepository.GetAcceptedForUserAsync(currentUserId);

            List<Friendship> friendshipList = friendships.ToList();

            List<string> friendUserIds = friendshipList
                .Select(f =>
                    f.RequesterUserId == currentUserId
                        ? f.AddresseeUserId
                        : f.RequesterUserId)
                .Distinct()
                .ToList();

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(friendUserIds);

            return friendUserIds.Select(userId =>
                new ChatFriendView
                {
                    UserId = userId,
                    Username = usernames.TryGetValue(
                        userId,
                        out string? username)
                            ? username
                            : "Unknown user"
                });
        }

        public async Task<IEnumerable<ChatFriendView>> GetAvailableFriendsToInviteAsync(int roomId)
        {
            string currentUserId = currentUser.GetRequiredUserId();

            ChatRoom room = await GetRoomOrThrowAsync(roomId);

            if (room.RoomType != ChatRoomType.Private)
                throw new InvalidOperationException(
                    "Friends can only be invited to private rooms.");

            await EnsureMembershipAsync(roomId, currentUserId);

            IEnumerable<Friendship> friendships =
                await friendshipRepository.GetAcceptedForUserAsync(currentUserId);

            List<Friendship> friendshipList = friendships.ToList();

            List<string> friendUserIds = friendshipList
                .Select(f =>
                    f.RequesterUserId == currentUserId
                        ? f.AddresseeUserId
                        : f.RequesterUserId)
                .Distinct()
                .ToList();

            IEnumerable<ChatRoomMember> members =
                await chatRepository.GetMembersAsync(roomId);

            HashSet<string> existingMemberIds = members
                .Select(m => m.UserId)
                .ToHashSet();

            List<string> availableFriendIds = friendUserIds
                .Where(id => !existingMemberIds.Contains(id))
                .ToList();

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(availableFriendIds);

            return availableFriendIds.Select(userId =>
                new ChatFriendView
                {
                    UserId = userId,
                    Username = usernames.TryGetValue(
                        userId,
                        out string? username)
                            ? username
                            : "Unknown user"
                });
        }

        public async Task<int> CreatePrivateRoomAsync(
            string name,
            IEnumerable<string> invitedUserIds)
        {
            string currentUserId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(name))
                throw new ArgumentException("A room name is required.");

            name = name.Trim();

            if (name.Length > 100)
                throw new ArgumentException(
                    "Room names cannot exceed 100 characters.");

            if (invitedUserIds == null)
                throw new ArgumentNullException(nameof(invitedUserIds));

            List<string> userIds = invitedUserIds
                .Where(id => !string.IsNullOrWhiteSpace(id))
                .Select(id => id.Trim())
                .Where(id => id != currentUserId)
                .Distinct()
                .ToList();

            foreach (string invitedUserId in userIds)
            {
                await EnsureAcceptedFriendshipAsync(
                    currentUserId,
                    invitedUserId);
            }

            ChatRoom room = new ChatRoom
            {
                Name = name,
                RoomType = ChatRoomType.Private,
                CreatedByUserId = currentUserId,
                OwnerUserId = currentUserId,
                CreatedAt = DateTime.UtcNow
            };

            List<ChatRoomMember> members = new()
            {
                new ChatRoomMember
                {
                    UserId = currentUserId,
                    JoinedAt = DateTime.UtcNow
                }
            };

            members.AddRange(userIds.Select(userId =>
                new ChatRoomMember
                {
                    UserId = userId,
                    JoinedAt = DateTime.UtcNow
                }));

            await chatRepository.CreateRoomWithMembersAsync(
                room,
                members);

            return room.Id;
        }

        public async Task InviteFriendAsync(
            int roomId,
            string invitedUserId)
        {
            string currentUserId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(invitedUserId))
                throw new ArgumentException("A user ID is required.");

            if (invitedUserId == currentUserId)
                throw new InvalidOperationException(
                    "You cannot invite yourself.");

            ChatRoom room = await GetRoomOrThrowAsync(roomId);

            if (room.RoomType != ChatRoomType.Private)
                throw new InvalidOperationException(
                    "Friends can only be invited to private rooms.");

            await EnsureMembershipAsync(roomId, currentUserId);

            await EnsureAcceptedFriendshipAsync(
                currentUserId,
                invitedUserId);

            ChatRoomMember? existingMember =
                await chatRepository.GetMembershipAsync(
                    roomId,
                    invitedUserId);

            if (existingMember != null)
                throw new InvalidOperationException(
                    "This user is already a member of the room.");

            await chatRepository.AddMemberAsync(new ChatRoomMember
            {
                ChatRoomId = roomId,
                UserId = invitedUserId,
                JoinedAt = DateTime.UtcNow
            });
        }

        // Membership
        // --------------------------------

        public async Task JoinMiscellaneousRoomAsync(int roomId)
        {
            string userId = currentUser.GetRequiredUserId();

            ChatRoom room = await GetRoomOrThrowAsync(roomId);

            if (room.RoomType != ChatRoomType.Miscellaneous)
                throw new InvalidOperationException(
                    "This room cannot be joined through this operation.");

            ChatRoomMember? existingMember =
                await chatRepository.GetMembershipAsync(roomId, userId);

            if (existingMember != null)
                return;

            await chatRepository.AddMemberAsync(new ChatRoomMember
            {
                ChatRoomId = roomId,
                UserId = userId,
                JoinedAt = DateTime.UtcNow
            });
        }

        public async Task RemoveTemporaryMembershipAsync(
    int roomId,
    string userId)
        {
            if (string.IsNullOrWhiteSpace(userId))
                return;

            // The room might have been deleted while the user was disconnected.
            ChatRoom? room =
                await chatRepository.GetRoomByIdAsync(roomId);

            if (room == null)
                return;

            // Never automatically remove private-room membership.
            if (room.RoomType != ChatRoomType.Miscellaneous &&
                room.RoomType != ChatRoomType.Topic)
            {
                return;
            }

            ChatRoomMember? membership =
                await chatRepository.GetMembershipAsync(roomId, userId);

            if (membership == null)
                return;

            await chatRepository.RemoveMemberAsync(membership);
        }

        public async Task LeaveRoomAsync(int roomId)
        {
            string userId = currentUser.GetRequiredUserId();

            await chatRepository.LeaveRoomAsync(roomId, userId);
        }



        // --------------------------------
        // Topic rooms
        // --------------------------------

        public async Task<int> GetOrCreateTopicRoomAsync(int topicId)
        {
            string userId = currentUser.GetRequiredUserId();

            if (topicId <= 0)
                throw new ArgumentException("Invalid topic ID.");

            bool topicExists =
                await chatRepository.TopicExistsAsync(topicId);

            if (!topicExists)
                throw new KeyNotFoundException("Topic not found.");

            ChatRoom? room =
                await chatRepository.GetRoomByTopicIdAsync(topicId);

            if (room == null)
            {
                string? topicTitle =
                    await chatRepository.GetTopicTitleAsync(topicId);

                room = new ChatRoom
                {
                    Name = topicTitle ?? $"Topic {topicId}",
                    RoomType = ChatRoomType.Topic,
                    CreatedByUserId = userId,
                    OwnerUserId = null,
                    TopicId = topicId,
                    CreatedAt = DateTime.UtcNow
                };

                try
                {
                    await chatRepository.CreateRoomWithMembersAsync(
                        room,
                        new[]
                        {
                            new ChatRoomMember
                            {
                                UserId = userId,
                                JoinedAt = DateTime.UtcNow
                            }
                        });
                }
                catch (Microsoft.EntityFrameworkCore.DbUpdateException)
                {
                    // Another request may have created this topic room.
                    room = await chatRepository.GetRoomByTopicIdAsync(topicId);

                    if (room == null)
                        throw;
                }
            }

            await GetRoomAsync(room.Id);

            return room.Id;
        }

        // --------------------------------
        // Messages
        // --------------------------------

        public async Task<IEnumerable<ChatMessageView>>
            GetLatestMessagesAsync(int roomId)
        {
            string userId = currentUser.GetRequiredUserId();

            await GetRoomAsync(roomId);

            IEnumerable<ChatMessage> messages =
                await chatRepository.GetLatestMessagesAsync(
                    roomId,
                    MessageLimit);

            List<ChatMessage> messageList = messages.ToList();

            Dictionary<string, string> usernames =
                await userService.GetUsernamesByIdsAsync(
                    messageList.Select(m => m.SenderUserId));

            return messageList.Select(message =>
                new ChatMessageView
                {
                    Id = message.Id,
                    ChatRoomId = message.ChatRoomId,
                    SenderUserId = message.SenderUserId,
                    SenderUsername =
                        usernames.TryGetValue(
                            message.SenderUserId,
                            out string? username)
                            ? username
                            : "Unknown user",
                    Content = message.Content,
                    SentAt = message.SentAt,
                    IsOwnMessage = message.SenderUserId == userId
                });
        }

        public async Task<ChatMessageView> SendMessageAsync(
            int roomId,
            string content)
        {
            string userId = currentUser.GetRequiredUserId();

            if (string.IsNullOrWhiteSpace(content))
                throw new ArgumentException(
                    "A message cannot be empty.");

            content = content.Trim();

            if (content.Length > MaxMessageLength)
                throw new ArgumentException(
                    $"Messages cannot exceed {MaxMessageLength} characters.");

            await GetRoomOrThrowAsync(roomId);
            await EnsureMembershipAsync(roomId, userId);

            ChatMessage message = new ChatMessage
            {
                ChatRoomId = roomId,
                SenderUserId = userId,
                Content = content,
                SentAt = DateTime.UtcNow
            };

            string username = await userService.GetUsernameByIdAsync(userId);

            await chatRepository.AddMessageAsync(message);

            return new ChatMessageView
            {
                Id = message.Id,
                ChatRoomId = message.ChatRoomId,
                SenderUserId = userId,
                SenderUsername = username,
                Content = message.Content,
                SentAt = message.SentAt,
                IsOwnMessage = true
            };
        }

        // --------------------------------
        // Private helper methods
        // --------------------------------

        private async Task<ChatRoom> GetRoomOrThrowAsync(int roomId)
        {
            ChatRoom? room =
                await chatRepository.GetRoomByIdAsync(roomId);

            if (room == null)
                throw new KeyNotFoundException("Chat room not found.");

            return room;
        }

        private async Task EnsureMembershipAsync(
            int roomId,
            string userId)
        {
            ChatRoomMember? membership =
                await chatRepository.GetMembershipAsync(roomId, userId);

            if (membership == null)
                throw new UnauthorizedAccessException(
                    "You are not a member of this chatroom.");
        }

        private async Task EnsureAcceptedFriendshipAsync(
            string firstUserId,
            string secondUserId)
        {
            Friendship? friendship =
                await friendshipRepository.GetBetweenUsersAsync(
                    firstUserId,
                    secondUserId);

            if (friendship == null ||
                friendship.Status != FriendshipStatus.Accepted)
            {
                throw new InvalidOperationException(
                    "You can only invite accepted friends.");
            }
        }

        private static ChatRoomView MapRoom(
            ChatRoom room,
            string currentUserId)
        {
            return new ChatRoomView
            {
                Id = room.Id,
                Name = room.Name,
                RoomType = room.RoomType.ToString(),
                OwnerUserId = room.OwnerUserId,
                IsOwner = room.OwnerUserId == currentUserId,
                IsMember = true,
                MemberCount = room.Members.Count,
                TopicId = room.TopicId
            };
        }
    }

                    ClaimTypes.NameIdentifier);

                if (string.IsNullOrEmpty(userId))
                    return Forbid();

                // First remove the user's database membership.
                await chatService.LeaveRoomAsync(roomId);

                // Find every active SignalR connection belonging to this user.
                IReadOnlyList<string> connections =
                    connectionTracker.GetConnections(userId);

                string groupName = ChatHub.GetGroupName(roomId);

                foreach (string connectionId in connections)

                    await chatHub.Groups.RemoveFromGroupAsync(
                        connectionId,
                        groupName);

                    // Also update our connection tracker.
                    connectionTracker.RemoveRoom(
                        connectionId,
                        roomId);
                }

                return Ok();
            }
            catch (KeyNotFoundException)
            {
                return NotFound();
            }
            catch (UnauthorizedAccessException)
            {

namespace OpenForum.Hubs
{
    [Authorize]
    public class ChatHub : Hub
    {
        private readonly IChatService chatService;
        private readonly ChatConnectionTracker connectionTracker;

        public ChatHub(
            IChatService chatService,
            ChatConnectionTracker connectionTracker)
        {
            this.chatService = chatService;
            this.connectionTracker = connectionTracker;
        }

        public override async Task OnConnectedAsync()
        {
            string? userId = Context.UserIdentifier;

            if (!string.IsNullOrEmpty(userId))
            {
                connectionTracker.AddConnection(
                    userId,
                    Context.ConnectionId);
            }

            await base.OnConnectedAsync();
        }

        public override async Task OnDisconnectedAsync(Exception? exception)
        {
            string? userId = Context.UserIdentifier;

            if (!string.IsNullOrEmpty(userId))
            {
                // Atomically remove this connection and find rooms
                // where this was the user's last active connection.
                var roomsToCleanUp =
                    connectionTracker.RemoveConnectionAndGetLastRooms(
                        userId,
                        Context.ConnectionId);

                foreach (var roomId in roomsToCleanUp)
                {
                    // The service will only remove membership from
                    // Miscellaneous and Topic rooms.
                    await chatService.RemoveTemporaryMembershipAsync(
                        roomId,
                        userId);
                }
            }

            await base.OnDisconnectedAsync(exception);
        }

        public async Task JoinRoom(int roomId)
        {
            if (roomId <= 0)
                throw new ArgumentException("Invalid room ID.");

            await chatService.GetRoomAsync(roomId);

            connectionTracker.AddRoom(Context.ConnectionId, roomId);

            try
            {
                await Groups.AddToGroupAsync(
                    Context.ConnectionId,
                    GetGroupName(roomId));
            }
            catch
            {
                connectionTracker.RemoveRoom(Context.ConnectionId, roomId);
                throw;
            }
        }

        public async Task LeaveRoom(int roomId)
        {
            if (roomId <= 0)
                throw new ArgumentException("Invalid room ID.");

            await Groups.RemoveFromGroupAsync(
                Context.ConnectionId,
                GetGroupName(roomId));

            connectionTracker.RemoveRoom(
                Context.ConnectionId,
                roomId);
        }

        public async Task SendMessage(int roomId, string content)
        {
            if (roomId <= 0)
                throw new ArgumentException("Invalid room ID.");

            var message = await chatService.SendMessageAsync(
                roomId,
                content);

            await Clients.Group(GetGroupName(roomId))
                .SendAsync("ReceiveMessage", message);
        }

        public static string GetGroupName(int roomId)
        {
            return $"chatroom_{roomId}";
        }
    }
}



using System.Collections.Concurrent;

public class ChatConnectionTracker
{
    private readonly object _lock = new();

    // User ID -> connection IDs
    private readonly Dictionary<string, HashSet<string>> _userConnections = new();

    // Connection ID -> joined room IDs
    private readonly Dictionary<string, HashSet<int>> _connectionRooms = new();

    public List<string> GetConnections(string userId)
    {
        lock (_lock)
        {
            return _userConnections.TryGetValue(userId, out var connections)
                ? connections.ToList()
                : new List<string>();
        }
    }

    public void AddConnection(string userId, string connectionId)
    {
        lock (_lock)
        {
            if (!_userConnections.TryGetValue(userId, out var connections))
            {
                connections = new HashSet<string>();
                _userConnections[userId] = connections;
            }

            connections.Add(connectionId);

            if (!_connectionRooms.ContainsKey(connectionId))
            {
                _connectionRooms[connectionId] = new HashSet<int>();
            }
        }
    }

    public void RemoveConnection(string userId, string connectionId)
    {
        lock (_lock)
        {
            if (_userConnections.TryGetValue(userId, out var connections))
            {
                connections.Remove(connectionId);

                if (connections.Count == 0)
                {
                    _userConnections.Remove(userId);
                }
            }

            _connectionRooms.Remove(connectionId);
        }
    }

    public List<int> RemoveConnectionAndGetLastRooms(
    string userId,
    string connectionId)
    {
        lock (_lock)
        {
            var joinedRooms = _connectionRooms.TryGetValue(
                connectionId,
                out var rooms)
                    ? rooms.ToList()
                    : new List<int>();

            _connectionRooms.Remove(connectionId);

            if (!_userConnections.TryGetValue(userId, out var connections))
                return joinedRooms;

            connections.Remove(connectionId);

            if (connections.Count == 0)
            {
                _userConnections.Remove(userId);
                return joinedRooms;
            }

            // Only return rooms that no remaining connection for this user
            // is currently joined to.
            return joinedRooms
                .Where(roomId => !connections.Any(otherConnectionId =>
                    _connectionRooms.TryGetValue(
                        otherConnectionId,
                        out var otherRooms)
                    && otherRooms.Contains(roomId)))
                .ToList();
        }
    }

    public void AddRoom(string connectionId, int roomId)
    {
        lock (_lock)
        {
            if (!_connectionRooms.TryGetValue(connectionId, out var rooms))
            {
                rooms = new HashSet<int>();
                _connectionRooms[connectionId] = rooms;
            }

            rooms.Add(roomId);
        }
    }

    public void RemoveRoom(string connectionId, int roomId)
    {
        lock (_lock)
        {
            if (_connectionRooms.TryGetValue(connectionId, out var rooms))
            {
                rooms.Remove(roomId);
            }
        }
    }

    public List<int> GetRooms(string connectionId)
    {
        lock (_lock)
        {
            return _connectionRooms.TryGetValue(connectionId, out var rooms)
                ? rooms.ToList()
                : new List<int>();
        }
    }

    public bool HasOtherConnectionInRoom(
        string userId,
        string connectionId,
        int roomId)
    {
        lock (_lock)
        {
            if (!_userConnections.TryGetValue(userId, out var connections))
                return false;

            return connections
                .Where(id => id != connectionId)
                .Any(id =>
                    _connectionRooms.TryGetValue(id, out var rooms)
                    && rooms.Contains(roomId));
        }
    }
}

document.addEventListener("DOMContentLoaded", async () => {

    const roomId = window.chatRoomId;

    const messagesContainer =
        document.getElementById("messagesContainer");

    const messagesLoading =
        document.getElementById("messagesLoading");

    const messageForm =
        document.getElementById("messageForm");

    const messageInput =
        document.getElementById("messageInput");

    const sendMessageButton =
        document.getElementById("sendMessageButton");

    const errorElement =
        document.getElementById("chatError");

    const MAX_MESSAGES = 100;

    const messages = new Map();

    let connection = null;
    let isPageLeaving = false;

    // --------------------------------
    // Error handling
    // --------------------------------

    function showError(message) {
        errorElement.textContent = message;
        errorElement.classList.remove("d-none");
    }

    function clearError() {
        errorElement.textContent = "";
        errorElement.classList.add("d-none");
    }

    // --------------------------------
    // Message rendering
    // --------------------------------

    function renderMessages() {

        const sortedMessages = Array.from(messages.values())
            .sort((a, b) =>
                new Date(a.sentAt) - new Date(b.sentAt)
            );

        // Keep the client-side message limit bounded.
        while (sortedMessages.length > MAX_MESSAGES) {
            const oldestMessage = sortedMessages.shift();
            messages.delete(oldestMessage.id);
        }

        messagesContainer.replaceChildren();

        sortedMessages.forEach(message => {

            // Determine ownership for the currently logged-in user.
            const isOwnMessage =
                message.senderUserId === window.chatCurrentUserId;

            const messageElement = document.createElement("div");

            messageElement.className =
                isOwnMessage
                    ? "align-self-start bg-primary text-white rounded p-3"
                    : "align-self-end bg-light border rounded p-3";

            messageElement.style.maxWidth = "75%";

            const sender = document.createElement("strong");
            sender.className = "d-block small mb-1";
            sender.textContent = message.senderUsername;

            const content = document.createElement("div");
            content.style.whiteSpace = "pre-wrap";
            content.style.overflowWrap = "anywhere";
            content.textContent = message.content;

            const timestamp = document.createElement("small");

            timestamp.className =
                isOwnMessage
                    ? "d-block mt-2 text-white-50"
                    : "d-block mt-2 text-muted";

            const sentAt = new Date(message.sentAt);

            timestamp.textContent = sentAt.toLocaleString();

            messageElement.append(
                sender,
                content,
                timestamp
            );

            messagesContainer.appendChild(messageElement);
        });

        messagesContainer.scrollTop =
            messagesContainer.scrollHeight;
    }

    function addMessage(message) {

        // Prevent duplicates when history and SignalR overlap.
        if (messages.has(message.id))
            return;

        messages.set(message.id, message);

        renderMessages();
    }

    // --------------------------------
    // Load existing messages
    // --------------------------------

    async function loadMessages() {

        try {
            const response = await fetch(
                `/Chat/Messages?roomId=${roomId}`
            );

            if (!response.ok) {
                throw new Error("Could not load chat messages.");
            }

            const loadedMessages = await response.json();

            loadedMessages.forEach(message => {
                addMessage(message);
            });

            messagesLoading.classList.add("d-none");

        } catch (error) {

            if (!isPageLeaving) {
                showError(error.message);
                messagesLoading.classList.add("d-none");
            }
        }
    }

    // --------------------------------
    // SignalR connection
    // --------------------------------

    async function connectToChat() {

        // Do not create another connection while leaving the page.
        if (isPageLeaving)
            return;

        // Avoid creating a duplicate connection.
        if (connection &&
            connection.state !== signalR.HubConnectionState.Disconnected) {
            return;
        }

        connection = new signalR.HubConnectionBuilder()
            .withUrl("/chatHub")
            .withAutomaticReconnect()
            .build();

        // Receive messages from other users.
        connection.on("ReceiveMessage", message => {
            if (!isPageLeaving) {
                addMessage(message);
            }
        });

        connection.onreconnecting(() => {

            if (isPageLeaving)
                return;

            showError("Connection lost. Reconnecting...");
            sendMessageButton.disabled = true;
        });

        connection.onreconnected(async () => {

            if (isPageLeaving)
                return;

            clearError();

            try {
                await connection.invoke("JoinRoom", roomId);

                if (isPageLeaving)
                    return;

                await loadMessages();

                sendMessageButton.disabled = false;

            } catch (error) {

                if (!isPageLeaving) {
                    showError("Could not rejoin the chatroom.");
                    sendMessageButton.disabled = true;
                }
            }
        });

        connection.onclose(() => {

            if (isPageLeaving)
                return;

            showError(
                "Connection to the chatroom was lost. Please refresh the page."
            );

            sendMessageButton.disabled = true;
        });

        try {

            await connection.start();

            // The page may have started unloading while connecting.
            if (isPageLeaving) {
                await connection.stop();
                return;
            }

            await connection.invoke("JoinRoom", roomId);

            clearError();

        } catch (error) {

            if (!isPageLeaving) {
                showError("Could not connect to the chatroom.");
                sendMessageButton.disabled = true;
            }
        }
    }

    // --------------------------------
    // Page lifecycle
    // --------------------------------

    // Stop SignalR when the user navigates away or closes the page.
    // This allows the server's OnDisconnectedAsync to run.
    window.addEventListener("pagehide", () => {

        isPageLeaving = true;

        if (connection) {
            connection.stop().catch(() => {
                // The connection may already have stopped.
            });
        }
    });

    // If the browser restores the page from its back-forward cache,
    // establish a fresh SignalR connection.
    window.addEventListener("pageshow", async event => {

        if (!event.persisted)
            return;

        isPageLeaving = false;

        await connectToChat();

        if (connection &&
            connection.state === signalR.HubConnectionState.Connected) {

            await loadMessages();
            await loadRoomMembers();
        }
    });

    // --------------------------------
    // Send messages
    // --------------------------------

    messageForm.addEventListener("submit", async event => {

        event.preventDefault();

        if (isPageLeaving)
            return;

        clearError();

        const content = messageInput.value.trim();

        if (!content)
            return;

        if (content.length > 2000) {
            showError("Messages cannot exceed 2000 characters.");
            return;
        }

        if (!connection ||
            connection.state !== signalR.HubConnectionState.Connected) {

            showError("You are not connected to the chatroom.");
            return;
        }

        sendMessageButton.disabled = true;

        try {

            await connection.invoke(
                "SendMessage",
                roomId,
                content
            );

            messageInput.value = "";
            messageInput.focus();

        } catch (error) {

            showError("Could not send your message.");

        } finally {

            sendMessageButton.disabled =
                !connection ||
                connection.state !== signalR.HubConnectionState.Connected;
        }
    });

    // --------------------------------
    // Room members
    // --------------------------------

    async function loadRoomMembers() {

        const membersList =
            document.getElementById("roomMembersList");

        const loading =
            document.getElementById("roomMembersLoading");

        try {

            const response = await fetch(
                `/Chat/RoomMembers?roomId=${roomId}`
            );

            if (!response.ok) {
                throw new Error("Could not load room members.");
            }

            const members = await response.json();

            membersList.replaceChildren();

            members.forEach(member => {

                const badge = document.createElement("span");

                badge.className = "badge bg-light text-dark border p-2";

                badge.textContent = member.isOwner
                    ? `${member.username} (Owner)`
                    : member.username;

                membersList.appendChild(badge);
            });

            document.getElementById("roomMemberCount").textContent =
                members.length;

        } catch (error) {

            if (!isPageLeaving) {
                showError(error.message);
            }

        } finally {

            loading.classList.add("d-none");
        }
    }

    // --------------------------------
    // Invite friends
    // --------------------------------

    const inviteFriendsButton =
        document.getElementById("inviteFriendsButton");

    if (inviteFriendsButton) {

        const inviteModalElement =
            document.getElementById("inviteFriendsModal");

        const inviteModal =
            new bootstrap.Modal(inviteModalElement);

        const inviteFriendsList =
            document.getElementById("inviteFriendsList");

        const inviteFriendsError =
            document.getElementById("inviteFriendsError");

        const inviteFriendsSuccess =
            document.getElementById("inviteFriendsSuccess");

        async function loadAvailableFriends() {

            inviteFriendsList.replaceChildren();

            inviteFriendsError.classList.add("d-none");
            inviteFriendsSuccess.classList.add("d-none");

            document.getElementById("inviteFriendsLoading")
                .classList.remove("d-none");

            document.getElementById("inviteFriendsEmpty")
                .classList.add("d-none");

            try {

                const response = await fetch(
                    `/Chat/AvailableFriendsToInvite?roomId=${roomId}`
                );

                if (!response.ok) {
                    throw new Error("Could not load available friends.");
                }

                const friends = await response.json();

                document.getElementById("inviteFriendsLoading")
                    .classList.add("d-none");

                if (friends.length === 0) {

                    document.getElementById("inviteFriendsEmpty")
                        .classList.remove("d-none");

                    return;
                }

                friends.forEach(friend => {

                    const item = document.createElement("div");

                    item.className =
                        "list-group-item d-flex justify-content-between align-items-center";

                    const username = document.createElement("span");

                    username.textContent = friend.username;

                    const button = document.createElement("button");

                    button.type = "button";
                    button.className = "btn btn-primary btn-sm";
                    button.textContent = "Invite";

                    button.addEventListener("click", async () => {
                        await inviteFriend(friend.userId, button);
                    });

                    item.append(username, button);

                    inviteFriendsList.appendChild(item);
                });

            } catch (error) {

                document.getElementById("inviteFriendsLoading")
                    .classList.add("d-none");

                inviteFriendsError.textContent = error.message;
                inviteFriendsError.classList.remove("d-none");
            }
        }

        async function inviteFriend(userId, button) {

            inviteFriendsError.classList.add("d-none");
            inviteFriendsSuccess.classList.add("d-none");

            const token = document.querySelector(
                'input[name="__RequestVerificationToken"]'
            )?.value;

            button.disabled = true;

            try {

                const response = await fetch("/Chat/InviteFriend", {

                    method: "POST",

                    headers: {
                        "Content-Type": "application/x-www-form-urlencoded"
                    },

                    body: new URLSearchParams({
                        roomId,
                        invitedUserId: userId,
                        __RequestVerificationToken: token
                    })
                });

                if (!response.ok) {

                    const result = await response.json();

                    throw new Error(
                        result.error || "Could not invite friend."
                    );
                }

                inviteFriendsSuccess.textContent =
                    "Friend successfully invited.";

                inviteFriendsSuccess.classList.remove("d-none");

                await loadRoomMembers();
                await loadAvailableFriends();

                inviteFriendsSuccess.textContent =
                    "Friend successfully invited.";

                inviteFriendsSuccess.classList.remove("d-none");

            } catch (error) {

                inviteFriendsError.textContent = error.message;
                inviteFriendsError.classList.remove("d-none");

                button.disabled = false;
            }
        }

        inviteFriendsButton.addEventListener("click", async () => {

            await loadAvailableFriends();

            inviteModal.show();
        });
    }

    // --------------------------------
    // Leave room
    // --------------------------------

    const leaveRoomButton =
        document.getElementById("leaveRoomButton");

    if (leaveRoomButton) {

        leaveRoomButton.addEventListener("click", async () => {

            clearError();

            const confirmed = confirm(
                "Are you sure you want to leave this chatroom?"
            );

            if (!confirmed)
                return;

            const token = document.querySelector(
                'input[name="__RequestVerificationToken"]'
            )?.value;

            leaveRoomButton.disabled = true;

            try {

                const response = await fetch("/Chat/LeaveRoom", {

                    method: "POST",

                    headers: {
                        "Content-Type": "application/x-www-form-urlencoded"
                    },

                    body: new URLSearchParams({
                        roomId,
                        __RequestVerificationToken: token
                    })
                });

                if (!response.ok) {

                    let errorMessage = "Could not leave the chatroom.";

                    try {
                        const result = await response.json();

                        if (result.error)
                            errorMessage = result.error;

                    } catch {
                        // Keep the default error message.
                    }

                    throw new Error(errorMessage);
                }

                // Successfully left the room.
                // Redirect to the application front page.
                window.location.href = "/";

            } catch (error) {

                showError(error.message);

                leaveRoomButton.disabled = false;
            }
        });
    }

    // --------------------------------
    // Initialize chat
    // --------------------------------

    if (!roomId || roomId <= 0) {
        showError("Invalid chatroom.");
        return;
    }

    await connectToChat();

    if (connection &&
        connection.state === signalR.HubConnectionState.Connected) {

        await loadMessages();
        await loadRoomMembers();
    }

});

using Application.Interfaces.Services.Features;
using Microsoft.AspNetCore.Mvc;

namespace OpenForum.ViewComponents
{
    public class UnreadMessagesViewComponent : ViewComponent
    {
        private readonly IMessageService messageService;

        public UnreadMessagesViewComponent(
            IMessageService messageService)
        {
            this.messageService = messageService;
        }

        public async Task<IViewComponentResult> InvokeAsync()
        {
            int unreadCount =
                await messageService.GetUnreadCountAsync();

            return View(unreadCount);
        }
    }
}
